"""
Escalera 3, marea alta

H. Mroz Quispe.  Picture no. 3.

Declaration.  Service stair no. 3, port side, seen from the half-landing.  Well 3.90 m wide, a spine
wall 0.16 m between two flights of fourteen risers, 0.17 m by 0.29 m.  The left flight goes down to
the boat landing, which is 2.38 m below the lens floor and at this hour under 1.38 m of water.  The
right flight goes up to the offices.  Board-marked concrete, sea-green gloss dado at 1.05 m following

neither in the frame.  Tide staff, life ring, a 100 mm downpipe, a handrail each side, a cup.

Everything is in order.  Please check again.

Usage:  python escalera_3_marea_alta.py [scale=1.0] [out.png] [supersampling]
        scale 0.5 is a study; 1.0 is 2400 x 3000.

Built as signed-distance solids in metres, lit by calculation:
  lamps  - the sodium glass as a sphere, the battens as lines, sampled; penumbra widens with distance
  bounce - what the lamps light becomes small lamps (and each of those sends one more), with shadows
  water  - a plane; Fresnel mirror above, turbid below, the steps seen through it by refraction

@njit(**NJ)
def hash3(i, j, k):
    n = (i * 374761393 + j * 668265263 + k * 1274126177) & 0x7fffffff
    n = ((n ^ (n >> 13)) * 1274126177) & 0x7fffffff
    return ((n ^ (n >> 16)) & 0xffffff) / 16777216.0

@njit(**NJ)
def noise3(x, y, z):
    i = math.floor(x); j = math.floor(y); k = math.floor(z)
    u = x - i; v = y - j; w = z - k
    u = u * u * (3 - 2 * u); v = v * v * (3 - 2 * v); w = w * w * (3 - 2 * w)
    i = int(i); j = int(j); k = int(k)
    a = hash3(i, j, k) + u * (hash3(i + 1, j, k) - hash3(i, j, k))
    b = hash3(i, j + 1, k) + u * (hash3(i + 1, j + 1, k) - hash3(i, j + 1, k))
    c = hash3(i, j, k + 1) + u * (hash3(i + 1, j, k + 1) - hash3(i, j, k + 1))
    d = hash3(i, j + 1, k + 1) + u * (hash3(i + 1, j + 1, k + 1) - hash3(i, j + 1, k + 1))
    e = a + v * (b - a); f = c + v * (d - c)
    return e + w * (f - e)

@njit(**NJ)
def fbm3(x, y, z, o):
    s = 0.0; a = 0.5; t = 0.0
    for i in range(o):
        s += a * noise3(x, y, z); t += a; a *= 0.5
        x = x * 2.03 + 11.3; y = y * 2.03 + 5.7; z = z * 2.03 + 3.1
    return s / t

@njit(**NJ)
def sstep(a, b, x):
    t = (x - a) / (b - a)
    if t < 0: t = 0.0
    if t > 1: t = 1.0
    return t * t * (3 - 2 * t)

@njit(**NJ)
def sd_box(x, y, z, bx, by, bz):
    qx = abs(x) - bx; qy = abs(y) - by; qz = abs(z) - bz
    ox = max(qx, 0.0); oy = max(qy, 0.0); oz = max(qz, 0.0)
    return math.sqrt(ox * ox + oy * oy + oz * oz) + min(max(qx, max(qy, qz)), 0.0)

@njit(**NJ)
def sd_box2(a, b, ha, hb):
    qa = abs(a) - ha; qb = abs(b) - hb
    oa = max(qa, 0.0); ob = max(qb, 0.0)
    return math.sqrt(oa * oa + ob * ob) + min(max(qa, qb), 0.0)

@njit(**NJ)
def quad(a, b):
    """distance to the quarter-plane a <= 0, b <= 0."""
    oa = max(a, 0.0); ob = max(b, 0.0)
    return math.sqrt(oa * oa + ob * ob) + min(max(a, b), 0.0)

@njit(**NJ)
def sd_cap(x, y, z, ax, ay, az, bx, by, bz, r):
    px = x - ax; py = y - ay; pz = z - az
    ex = bx - ax; ey = by - ay; ez = bz - az
    h = (px * ex + py * ey + pz * ez) / (ex * ex + ey * ey + ez * ez + 1e-12)
    if h < 0: h = 0.0
    if h > 1: h = 1.0
    px -= ex * h; py -= ey * h; pz -= ez * h
    return math.sqrt(px * px + py * py + pz * pz) - r

@njit(**NJ)
def tex(T, u, v):
    h = T.shape[0]; w = T.shape[1]
    if u < 0.0 or u >= 1.0 or v < 0.0 or v >= 1.0: return 0.0
    x = u * w - 0.5; y = v * h - 0.5
    i = int(math.floor(x)); j = int(math.floor(y)); fx = x - i; fy = y - j
    i0 = min(max(i, 0), w - 1); i1 = min(max(i + 1, 0), w - 1)
    j0 = min(max(j, 0), h - 1); j1 = min(max(j + 1, 0), h - 1)
    a = T[j0, i0] + fx * (T[j0, i1] - T[j0, i0]); b = T[j1, i0] + fx * (T[j1, i1] - T[j1, i0])
    return a + fy * (b - a)

@njit(**NJ)
def water_h(x, y):
    """height of a point above the surface of the water, measured square to that surface."""
    return (x - WX0) * WNX + (y - WY0) * WNY

@njit(**NJ)
def march(ox, oy, oz, dx, dy, dz, tmax):
    t = 0.0
    for s in range(600):
        d, m = scene(ox + dx * t, oy + dy * t, oz + dz * t)
        if d < 0.00012 * t + 0.00003:
            return t, m
        t += d
        if t > tmax: break
    return 1e9, 0

@njit(**NJ)
def trace(ox, oy, oz, dx, dy, dz, tmax):
    """the solids, and the surface of the water if it comes first."""
    t, m = march(ox, oy, oz, dx, dy, dz, tmax)
    h = water_h(ox, oy); den = dx * WNX + dy * WNY
    if h > 0.0 and den < -1e-9:
        tw = -h / den
        if tw < t: return tw, M_WATER
    return t, m

@njit(**NJ)
def occl(ox, oy, oz, dx, dy, dz, maxt):
    t = 0.0
    for s in range(300):
        d, m = scene(ox + dx * t, oy + dy * t, oz + dz * t)
        if d < 0.0003: return True
        t += d
        if t >= maxt: return False
    return False

@njit(**NJ)
def normal(px, py, pz, t):
    e = 0.00012 * t + 0.0001
    nx = scene(px + e, py, pz)[0] - scene(px - e, py, pz)[0]
    ny = scene(px, py + e, pz)[0] - scene(px, py - e, pz)[0]
    nz = scene(px, py, pz + e)[0] - scene(px, py, pz - e)[0]
    nl = math.sqrt(nx * nx + ny * ny + nz * nz) + 1e-12
    return nx / nl, ny / nl, nz / nl

@njit(**NJ)
def dado_line(x, y, z):
    """top of the green paint: 1.05 m over the floor on a landing, the same over the nosings on a flight."""
    if z > ZE:
        if y > YU - 0.1: return YU + DADO
        return YB + DADO
    if z < ZS + 0.002: return DADO
    if x < 0.0: return DADO - PITCH * (z - ZS)
    return DADO + PITCH * (z - ZS) + 0.10

                h2 = hw + 0.17 + 0.16 * j + 0.012 * (fbm3(tu * 3.0, 4.0 + j, 2.3, 2) - 0.5)
                ln = math.exp(-((py - h2) / 0.005) ** 2) * (0.3 + 0.7 * noise3(tu * 6.0, float(j), 1.0)) * tw
                ar += 0.07 * ln; ag += 0.07 * ln; ab += 0.065 * ln
        cr = math.exp(-((py - hw - 0.004) / 0.011) ** 2) * (0.35 + 0.65 * fbm3(tu * 40.0, py * 40.0, 4.0, 2)) * tw   # salt and shell at the line
        ar += (0.60 - ar) * 0.75 * cr; ag += (0.58 - ag) * 0.75 * cr; ab += (0.52 - ab) * 0.75 * cr
        ro = ro + (0.7 - ro) * cr
    if f0g == 0.0: f0g = f0r; f0b = f0r
    nl = math.sqrt(nx * nx + ny * ny + nz * nz)
    return ar, ag, ab, f0r, f0g, f0b, ro, nx / nl, ny / nl, nz / nl


# ------------------------------------------------------------------ the lamps, directly
@njit(**NJ)
def lamps_at(px, py, pz, nx, ny, nz, vx, vy, vz, ro, q, Q, rot):
    """light arriving straight from the three lamps (one sample of each), and the highlight they make
    on a surface of roughness ro seen along v.  Returns irradiance rgb and highlight rgb."""
    er = 0.0; eg = 0.0; eb = 0.0; sr = 0.0; sg = 0.0; sb = 0.0
    hp = water_h(px, py)
    ns = 2.0 / max(ro * ro, 1e-4) - 2.0
    for l in range(3):
        if l == 0:
            cx = L0X - px; cy = L0Y - py; cz = L0Z - pz
            dc = math.sqrt(cx * cx + cy * cy + cz * cz)
            if abs(cy) < 0.9 * dc: tx = -cz; ty = 0.0; tz = cx
            else: tx = 1.0; ty = 0.0; tz = 0.0
            tl = math.sqrt(tx * tx + ty * ty + tz * tz); tx /= tl; ty /= tl; tz /= tl
            bx = (cy * tz - cz * ty) / dc; by = (cz * tx - cx * tz) / dc; bz = (cx * ty - cy * tx) / dc
            rad = L0R * math.sqrt((q + 0.5) / Q); th = 2.39996323 * q + rot
            lx = cx + rad * (math.cos(th) * tx + math.sin(th) * bx); ly = cy + rad * (math.cos(th) * ty + math.sin(th) * by)
            lz = cz + rad * (math.cos(th) * tz + math.sin(th) * bz)
            d = math.sqrt(lx * lx + ly * ly + lz * lz); lx /= d; ly /= d; lz /= d
            cn = nx * lx + ny * ly + nz * lz
            if cn <= 0.0: continue
            if dc > 0.16 and occl(px, py, pz, lx, ly, lz, d - 0.135): continue
            g = cn / max(dc * dc, 0.01)
            cr = I0R * g; cg = I0G * g; cb = I0B * g
        else:
            if l == 1: ox = L1X; oy = L1Y; oz = L1Z; it = I1
            else: ox = L2X; oy = L2Y; oz = L2Z; it = I2
            if it <= 0.0: continue
            u = (((q * 7 + 3 * l) % Q) + (rot * 0.15915494 + 0.37 * l) % 1.0) / Q
            lx = ox + (u - 0.5) * TLEN - px; ly = oy - py; lz = oz - pz
            if ly <= 0.0: continue
            d2 = lx * lx + ly * ly + lz * lz; d = math.sqrt(d2); lx /= d; ly /= d; lz /= d
            cn = nx * lx + ny * ly + nz * lz
            if cn <= 0.0: continue
            g = it * (0.15 + 0.85 * ly) * cn / d2
            if g < 2e-5: continue
            if occl(px, py, pz, lx, ly, lz, d - 0.03): continue
            cr = ITR * g; cg = ITG * g; cb = ITB * g
        if hp < 0.0:                                         # the point is under water: the light has come down through it
            pl = -hp / max(lx * WNX + ly * WNY, 0.2)
            cr *= 0.95 * math.exp(-SIG_R * pl); cg *= 0.95 * math.exp(-SIG_G * pl); cb *= 0.95 * math.exp(-SIG_B * pl)
        er += cr; eg += cg; eb += cb
        if ro < 0.6:
            hx = lx - vx; hy = ly - vy; hz = lz - vz; hl = math.sqrt(hx * hx + hy * hy + hz * hz) + 1e-12
            nh = max((nx * hx + ny * hy + nz * hz) / hl, 0.0)
            s = (ns + 8.0) / 8.0 * nh ** ns / math.pi
            sr += cr * s; sg += cg * s; sb += cb * s
    return er, eg, eb, sr, sg, sb


# ------------------------------------------------------------------ what the lamps light becomes small lamps
@njit(**NJ)
def make_lamps(n0, n1, n2):
    n = n0 + n1 + n2
    VP = np.zeros((2 * n, 3)); VN = np.zeros((2 * n, 3)); VC = np.zeros((2 * n, 3)); VR = np.full(2 * n, 0.40 ** 2)
    k = 0
    for i in range(n):
        if i < n0:
            cz_ = 1.0 - 2.0 * (i + 0.5) / n0; s_ = math.sqrt(1.0 - cz_ * cz_); ph = 2.39996323 * i
            dx = s_ * math.cos(ph); dy = cz_; dz = s_ * math.sin(ph)
            ox = L0X + dx * 0.105; oy = L0Y + dy * 0.105; oz = L0Z + dz * 0.105
            fr = 4 * math.pi * I0R / n0; fg = 4 * math.pi * I0G / n0; fb = 4 * math.pi * I0B / n0
        else:
            if i < n0 + n1: j = i - n0; nn = n1; cx = L1X; cy = L1Y; cz = L1Z; it = I1; l = 1
            else: j = i - n0 - n1; nn = n2; cx = L2X; cy = L2Y; cz = L2Z; it = I2; l = 2
            r1 = (j + hash3(j, 61, l)) / nn; r2 = hash3(j, 63, l) * 2 * math.pi
            r1 = (r1 * 37.0) % 1.0
            if j % 4 == 0: cd = r1
            else: cd = math.sqrt(r1)
            sd = math.sqrt(max(1.0 - cd * cd, 0.0))
            dx = sd * math.cos(r2); dy = -cd; dz = sd * math.sin(r2)
            ox = cx + ((j + 0.5) / nn - 0.5) * TLEN; oy = cy - 0.03; oz = cz
            f = it * math.pi * 1.15 / nn
            fr = ITR * f; fg = ITG * f; fb = ITB * f
        if scene(ox, oy, oz)[0] < 0.004 or fr + fg + fb <= 0.0: continue
        t, m = march(ox, oy, oz, dx, dy, dz, 40.0)
        if m == 0 or m == M_GLASS: continue
        px = ox + dx * t; py = oy + dy * t; pz = oz + dz * t
        hp = water_h(px, py)
        if hp < 0.0:
            pl = -hp / max(-(dx * WNX + dy * WNY), 0.2)
            fr *= 0.95 * math.exp(-SIG_R * pl); fg *= 0.95 * math.exp(-SIG_G * pl); fb *= 0.95 * math.exp(-SIG_B * pl)
        nx, ny, nz = normal(px, py, pz, 1.0)
        ar, ag, ab, f0r, f0g, f0b, ro, sx, sy, sz = material(px, py, pz, nx, ny, nz, m, 1.0)
        fr *= ar; fg *= ag; fb *= ab
        if fr + fg + fb < 1e-9: continue
        VP[k, 0] = px + nx * 0.01; VP[k, 1] = py + ny * 0.01; VP[k, 2] = pz + nz * 0.01
        VN[k, 0] = nx; VN[k, 1] = ny; VN[k, 2] = nz
        VC[k, 0] = fr / math.pi; VC[k, 1] = fg / math.pi; VC[k, 2] = fb / math.pi
        if t < 0.5: VR[k] = 0.25 ** 2
        k += 1
        # one more flight, cosine-distributed about the normal
        qx0 = VP[k - 1, 0]; qy0 = VP[k - 1, 1]; qz0 = VP[k - 1, 2]
        if abs(ny) < 0.9: tx = -nz; ty = 0.0; tz = nx
        else: tx = 1.0; ty = 0.0; tz = 0.0
        tl = math.sqrt(tx * tx + ty * ty + tz * tz); tx /= tl; ty /= tl; tz /= tl
        bx = ny * tz - nz * ty; by = nz * tx - nx * tz; bz = nx * ty - ny * tx
        r1 = hash3(i, 3, 41); r2 = hash3(i, 5, 41) * 2 * math.pi
        s_ = math.sqrt(r1); c = math.sqrt(1 - r1)
        dx = s_ * math.cos(r2) * tx + s_ * math.sin(r2) * bx + c * nx
        dy = s_ * math.cos(r2) * ty + s_ * math.sin(r2) * by + c * ny
        dz = s_ * math.cos(r2) * tz + s_ * math.sin(r2) * bz + c * nz
        t, m = march(qx0, qy0, qz0, dx, dy, dz, 40.0)
        if m == 0 or m == M_GLASS: continue
        qx = qx0 + dx * t; qy = qy0 + dy * t; qz = qz0 + dz * t
        if water_h(qx, qy) < 0.0 or hp < 0.0: continue
        nx, ny, nz = normal(qx, qy, qz, 1.0)
        ar, ag, ab, f0r, f0g, f0b, ro, sx, sy, sz = material(qx, qy, qz, nx, ny, nz, m, 1.0)
        VP[k, 0] = qx + nx * 0.01; VP[k, 1] = qy + ny * 0.01; VP[k, 2] = qz + nz * 0.01
        VN[k, 0] = nx; VN[k, 1] = ny; VN[k, 2] = nz
        VC[k, 0] = fr * ar / math.pi; VC[k, 1] = fg * ag / math.pi; VC[k, 2] = fb * ab / math.pi
        VR[k] = 0.9 ** 2
        k += 1
    return VP[:k], VN[:k], VC[:k], VR[:k]


@njit(**NJ)
def irradiance(px, py, pz, nx, ny, nz, VP, VN, VC, VR):
    """light arriving at a point from the small lamps, with shadows."""
    er = 0.0; eg = 0.0; eb = 0.0
    qx = px + nx * 0.004; qy = py + ny * 0.004; qz = pz + nz * 0.004
    for i in range(VP.shape[0]):
        lx = VP[i, 0] - qx; ly = VP[i, 1] - qy; lz = VP[i, 2] - qz
        r2 = lx * lx + ly * ly + lz * lz; r = math.sqrt(r2)
        cx = (nx * lx + ny * ly + nz * lz) / r
        if cx <= 0.0: continue
        ci = -(VN[i, 0] * lx + VN[i, 1] * ly + VN[i, 2] * lz) / r
        if ci <= 0.0: continue
        g = ci * cx / max(r2, VR[i])
        if g * (VC[i, 0] + VC[i, 1] + VC[i, 2]) < 2e-6: continue
        if occl(qx, qy, qz, lx / r, ly / r, lz / r, r - 0.012): continue
        er += VC[i, 0] * g; eg += VC[i, 1] * g; eb += VC[i, 2] * g
    hp = water_h(px, py)
    if hp < 0.0:
        er *= math.exp(1.5 * SIG_R * hp); eg *= math.exp(1.5 * SIG_G * hp); eb *= math.exp(1.5 * SIG_B * hp)
    return er, eg, eb


GNX, GNY, GNZ = 12, 20, 26
GX0, GY0, GZ0 = XL + 0.12, YB + 0.12, ZF + 0.12
GDX, GDY, GDZ = (XR - XL - 0.24) / (GNX - 1), (YT - YB - 0.24) / (GNY - 1), (ZB - ZF - 0.24) / (GNZ - 1)

@njit(parallel=True, **NJ)
def make_grid(VP, VN, VC, VR):
    """a coarse lattice of the bounced light, for six facings; used for what is seen in reflections."""
    G = np.zeros((GNX, GNY, GNZ, 6, 3)); GV = np.zeros((GNX, GNY, GNZ))
    for i in prange(GNX):
        for j in range(GNY):
            for k in range(GNZ):
                x = GX0 + i * GDX; y = GY0 + j * GDY; z = GZ0 + k * GDZ
                if scene(x, y, z)[0] < 0.03: continue
                GV[i, j, k] = 1.0
                for f in range(6):
                    nx = 0.0; ny = 0.0; nz = 0.0
                    if f == 0: nx = 1.0
                    elif f == 1: nx = -1.0
                    elif f == 2: ny = 1.0
                    elif f == 3: ny = -1.0
                    elif f == 4: nz = 1.0
                    else: nz = -1.0
                    r, g, b = irradiance(x, y, z, nx, ny, nz, VP, VN, VC, VR)
                    G[i, j, k, f, 0] = r; G[i, j, k, f, 1] = g; G[i, j, k, f, 2] = b
    return G, GV

@njit(**NJ)
def grid_irr(G, GV, x, y, z, nx, ny, nz):
    x += nx * 0.15; y += ny * 0.15; z += nz * 0.15
    fx = min(max((x - GX0) / GDX, 0.0), GNX - 1.001); fy = min(max((y - GY0) / GDY, 0.0), GNY - 1.001)
    fz = min(max((z - GZ0) / GDZ, 0.0), GNZ - 1.001)
    i = int(fx); j = int(fy); k = int(fz); fx -= i; fy -= j; fz -= k
    a0 = 0 if nx > 0 else 1; a1 = 2 if ny > 0 else 3; a2 = 4 if nz > 0 else 5
    w0 = nx * nx; w1 = ny * ny; w2 = nz * nz
    r = 0.0; g = 0.0; b = 0.0; ws = 0.0
    for di in range(2):
        wi = fx if di == 1 else 1 - fx
        for dj in range(2):
            wj = fy if dj == 1 else 1 - fy
            for dk in range(2):
                wk = fz if dk == 1 else 1 - fz
                w = wi * wj * wk * GV[i + di, j + dj, k + dk]
                if w <= 0.0: continue
                c = G[i + di, j + dj, k + dk]
                r += w * (w0 * c[a0, 0] + w1 * c[a1, 0] + w2 * c[a2, 0])
                g += w * (w0 * c[a0, 1] + w1 * c[a1, 1] + w2 * c[a2, 1])
                b += w * (w0 * c[a0, 2] + w1 * c[a1, 2] + w2 * c[a2, 2])
                ws += w
    if ws < 1e-4: return 0.0, 0.0, 0.0
    return r / ws, g / ws, b / ws


# ------------------------------------------------------------------ the water as an optical surface
@njit(**NJ)
def water_normal(x, z):
    """it is still, not polished: the surface moves by a few tenths of a millimetre."""
    e = 0.02
    h0 = fbm3(x * 1.4, 2.0, z * 1.4, 2)
    gx = (fbm3((x + e) * 1.4, 2.0, z * 1.4, 2) - h0) / e; gz = (fbm3(x * 1.4, 2.0, (z + e) * 1.4, 2) - h0) / e
    a = 0.0016
    nx = WNX - a * gx; ny = WNY; nz = -a * gz
    l = math.sqrt(nx * nx + ny * ny + nz * nz)
    return nx / l, ny / l, nz / l

@njit(**NJ)
def fresnel(c):
    n = 1.333
    s2 = (1.0 - c * c) / (n * n)
    if s2 >= 1.0: return 1.0
    ct = math.sqrt(1.0 - s2)
    rs = (c - n * ct) / (c + n * ct); rp = (n * c - ct) / (n * c + ct)
    return 0.5 * (rs * rs + rp * rp)

@njit(**NJ)
def refract(dx, dy, dz, nx, ny, nz):
    c = -(dx * nx + dy * ny + dz * nz); eta = 1.0 / 1.333
    k = 1.0 - eta * eta * (1.0 - c * c)
    a = eta * c - math.sqrt(max(k, 0.0))
    return eta * dx + a * nx, eta * dy + a * ny, eta * dz + a * nz


# ------------------------------------------------------------------ the bounce, measured on a coarser raster
@njit(**NJ)
def cam_ray(fx, fy, W, H):
    u = (fx / W * 2.0 - 1.0) * TU
    v = VTOP - fy / H * (VTOP - VBOT)
    l = math.sqrt(u * u + v * v + 1.0)
    return u / l, v / l, 1.0 / l

@njit(parallel=True, **NJ)
def bounce_pass(W, H, nx_, ny_, step, VP, VN, VC, VR, layer):
    """layer 0: what the lens sees.  layer 1: what it sees mirrored in the water.  layer 2: what it sees through the water."""
    BP = np.zeros((ny_, nx_, 3), np.float32); BN = np.zeros((ny_, nx_, 3), np.float32)
    BE = np.zeros((ny_, nx_, 3), np.float32)
    for iy in prange(ny_):
        for ix in range(nx_):
            BP[iy, ix, 0] = 1e6
            dx, dy, dz = cam_ray((ix + 0.5) * step, (iy + 0.5) * step, W, H)
            t, m = trace(CAMX, CAMY, 0.0, dx, dy, dz, 40.0)
            if m == 0 or m == M_GLASS: continue
            px = CAMX + dx * t; py = CAMY + dy * t; pz = dz * t
            if layer == 0:
                if m == M_WATER: continue
            else:
                if m != M_WATER: continue
                if layer == 1:
                    c = dx * WNX + dy * WNY
                    dx -= 2 * c * WNX; dy -= 2 * c * WNY
                    px += WNX * 0.003; py += WNY * 0.003
                else:
                    dx, dy, dz = refract(dx, dy, dz, WNX, WNY, 0.0)
                    px -= WNX * 0.003; py -= WNY * 0.003
                t2, m = march(px, py, pz, dx, dy, dz, 30.0)
                if m == 0 or m == M_GLASS: continue
                px += dx * t2; py += dy * t2; pz += dz * t2; t += t2
            nx, ny, nz = normal(px, py, pz, t)
            r, g, b = irradiance(px, py, pz, nx, ny, nz, VP, VN, VC, VR)
            BP[iy, ix, 0] = px; BP[iy, ix, 1] = py; BP[iy, ix, 2] = pz
            BN[iy, ix, 0] = nx; BN[iy, ix, 1] = ny; BN[iy, ix, 2] = nz
            BE[iy, ix, 0] = r; BE[iy, ix, 1] = g; BE[iy, ix, 2] = b
    return BP, BN, BE

@njit(**NJ)
def bounce_at(BP, BN, BE, fx, fy, px, py, pz, nx, ny, nz, t):
    """bounced light at a full-size sample, from the coarse raster, taking only neighbours on the same surface."""
    h, w = BE.shape[0], BE.shape[1]
    x0 = int(math.floor(fx - 0.5)); y0 = int(math.floor(fy - 0.5))
    sd = 0.0015 + 0.0012 * t
    r = 0.0; g = 0.0; b = 0.0; ws = 0.0
    for a in range(-1, 3):
        yy = y0 + a
        if yy < 0 or yy >= h: continue
        for c in range(-1, 3):
            xx = x0 + c
            if xx < 0 or xx >= w: continue
            if BP[yy, xx, 0] > 1e5: continue
            nn = nx * BN[yy, xx, 0] + ny * BN[yy, xx, 1] + nz * BN[yy, xx, 2]
            if nn < 0.85: continue
            pd = abs((BP[yy, xx, 0] - px) * nx + (BP[yy, xx, 1] - py) * ny + (BP[yy, xx, 2] - pz) * nz)
            ddx = xx + 0.5 - fx; ddy = yy + 0.5 - fy
            wg = math.exp(-(pd / sd) ** 2) * math.exp(-(ddx * ddx + ddy * ddy) / 0.7) * nn ** 8
            r += wg * BE[yy, xx, 0]; g += wg * BE[yy, xx, 1]; b += wg * BE[yy, xx, 2]; ws += wg
    if ws < 1e-3:
        return False, 0.0, 0.0, 0.0
    return True, r / ws, g / ws, b / ws


# ------------------------------------------------------------------ the exposure
@njit(**NJ)
def plain(ox, oy, oz, dx, dy, dz, tt, G, GV, AMB, emit, q, Q, rot, pa):
    """what a glossy surface sees along one reflected ray: lit by the lamps and the lattice, no further gloss."""
    t, m = trace(ox, oy, oz, dx, dy, dz, 40.0)
    if m == 0: return 0.0, 0.0, 0.0
    px = ox + dx * t; py = oy + dy * t; pz = oz + dz * t
    k = 1.0
    if m == M_WATER:                                         # water seen in paint: follow its mirror once
        c = -(dx * WNX + dy * WNY); k = fresnel(c)
        dx += 2 * c * WNX; dy += 2 * c * WNY
        px += WNX * 0.003; py += WNY * 0.003
        t2, m = march(px, py, pz, dx, dy, dz, 30.0)
        if m == 0: return 0.0, 0.0, 0.0
        px += dx * t2; py += dy * t2; pz += dz * t2; t += t2
    if m == M_GLASS:
        if emit: return k * LGR, k * LGG, k * LGB
        return 0.0, 0.0, 0.0
    hx, hy, hz = normal(px, py, pz, tt + t)
    ar, ag, ab, f0r, f0g, f0b, ro, nx, ny, nz = material(px, py, pz, hx, hy, hz, m, (tt + t) * pa * 3.0)
    er, eg, eb, sr, sg, sb = lamps_at(px + hx * 0.003, py + hy * 0.003, pz + hz * 0.003, nx, ny, nz, dx, dy, dz, 1.0, q, Q, rot)
    ir, ig, ib = grid_irr(G, GV, px, py, pz, hx, hy, hz)
    if f0r > 0.5:
        return k * f0r * (er + ir) / math.pi, k * f0g * (eg + ig) / math.pi, k * f0b * (eb + ib) / math.pi
    return k * ar * (er + ir + AMB[0]) / math.pi, k * ag * (eg + ig + AMB[1]) / math.pi, k * ab * (eb + ib + AMB[2]) / math.pi


@njit(**NJ)
def second(px, py, pz, t, m, dx, dy, dz, fx, fy, BP, BN, BE, G, GV, AMB, q, Q, rot, pa):
    """a surface met after the water: mirrored in it, or lying under it."""
    if m == 0: return 0.0, 0.0, 0.0
    if m == M_GLASS: return LGR, LGG, LGB
    gx, gy, gz = normal(px, py, pz, t)
    ar, ag, ab, f0r, f0g, f0b, ro, nx, ny, nz = material(px, py, pz, gx, gy, gz, m, t * pa * 1.5)
    er, eg, eb, sr, sg, sb = lamps_at(px + gx * 0.003, py + gy * 0.003, pz + gz * 0.003, nx, ny, nz, dx, dy, dz, ro, q, Q, rot)
    ok, ir, ig, ib = bounce_at(BP, BN, BE, fx, fy, px, py, pz, gx, gy, gz, t)
    if not ok:
        ir, ig, ib = grid_irr(G, GV, px, py, pz, gx, gy, gz)
    if f0r > 0.5:
        return f0r * (er + ir) / math.pi + f0r * sr, f0g * (eg + ig) / math.pi + f0g * sg, f0b * (eb + ib) / math.pi + f0b * sb
    lr = ar * (er + ir + AMB[0]) / math.pi; lg = ag * (eg + ig + AMB[1]) / math.pi; lb = ab * (eb + ib + AMB[2]) / math.pi
    if ro < 0.6 and f0r > 0.0:
        c1 = 1.0 - min(max(-(dx * nx + dy * ny + dz * nz), 0.0), 1.0)
        F = f0r + (1 - f0r) * c1 ** 5 * max(1.0 - 1.5 * ro, 0.0)
        lr = lr * (1 - F) + F * sr; lg = lg * (1 - F) + F * sg; lb = lb * (1 - F) + F * sb
    return lr, lg, lb


@njit(parallel=True, **NJ)
def expose(W, H, SS, step, BP, BN, BE, RP, RN, RE, UP, UN, UE, G, GV, AMB, seed):
    img = np.zeros((H, W, 3), np.float32)
    N = SS * SS
    pa = 2.0 * TU / W
    for iy in prange(H):
        for ix in range(W):
            pr = hash3(ix, iy, seed) * 6.2831853; pr2 = hash3(ix, iy, seed + 7) * 6.2831853
            sh = int(hash3(ix, iy, seed + 3) * N)
            cr = 0.0; cg = 0.0; cb = 0.0
            for s in range(N):
                sa = s // SS; sb_ = s % SS
                fx = ix + (sa + hash3(ix * 4 + sa, iy * 4 + sb_, seed + 11)) / SS
                fy = iy + (sb_ + hash3(ix * 4 + sa, iy * 4 + sb_, seed + 13)) / SS
                dx, dy, dz = cam_ray(fx, fy, W, H)
                t, m = trace(CAMX, CAMY, 0.0, dx, dy, dz, 40.0)
                if m == 0: continue
                px = CAMX + dx * t; py = CAMY + dy * t; pz = dz * t
                q = (s * 7 + sh) % N
                if m == M_GLASS:
                    cr += LGR; cg += LGG; cb += LGB
                    continue
                if m == M_WATER:
                    wx, wy, wz = water_normal(px, pz)
                    c = -(dx * wx + dy * wy + dz * wz)
                    F = fresnel(c)
                    # mirrored
                    rx = dx + 2 * c * wx; ry = dy + 2 * c * wy; rz = dz + 2 * c * wz
                    ox = px + WNX * 0.003; oy = py + WNY * 0.003
                    t2, m2 = march(ox, oy, pz, rx, ry, rz, 30.0)
                    ar_, ag_, ab_ = second(ox + rx * t2, oy + ry * t2, pz + rz * t2, t + t2, m2, rx, ry, rz, fx / step, fy / step,
                                           RP, RN, RE, G, GV, AMB, q, N, pr, pa)
                    # seen through
                    tx, ty, tz = refract(dx, dy, dz, wx, wy, wz)
                    ox = px - WNX * 0.003; oy = py - WNY * 0.003
                    t3, m3 = march(ox, oy, pz, tx, ty, tz, 12.0)
                    er, eg, eb, sr, sg, sb = lamps_at(px + WNX * 0.004, py + WNY * 0.004, pz, WNX, WNY, 0.0, dx, dy, dz, 1.0, q, N, pr)
                    ir, ig, ib = grid_irr(G, GV, px, py + 0.1, pz, 0.0, 1.0, 0.0)
                    er += ir; eg += ig; eb += ib
                    if m3 == 0:
                        t3 = 12.0; br = 0.0; bg = 0.0; bb = 0.0
                    else:
                        br, bg, bb = second(ox + tx * t3, oy + ty * t3, pz + tz * t3, t + t3, m3, tx, ty, tz, fx / step, fy / step,
                                            UP, UN, UE, G, GV, AMB, q, N, pr, pa)
                    a1 = math.exp(-SIG_R * t3); a2 = math.exp(-SIG_G * t3); a3 = math.exp(-SIG_B * t3)
                    br = br * a1 + 0.020 * er / math.pi * (1 - a1)           # what is in the water scatters a little of the lamp back
                    bg = bg * a2 + 0.030 * eg / math.pi * (1 - a2)
                    bb = bb * a3 + 0.022 * eb / math.pi * (1 - a3)
                    lr = F * ar_ + (1 - F) * br; lg = F * ag_ + (1 - F) * bg; lb = F * ab_ + (1 - F) * bb
                    # what floats: a skin of dust gathered along the walls, a few specks, the wetted line itself
                    ds = scene(px, py - 0.004, pz)[0]
                    sc = 0.30 * math.exp(-ds / 0.20) * sstep(0.45, 0.75, fbm3(px * 3.1, 5.0, pz * 3.1, 3))
                    sc += 0.30 * math.exp(-ds / 0.012) * (0.4 + 0.6 * noise3(px * 31.0, 1.0, pz * 31.0))
                    ci = int(math.floor(px / 0.012)); cj = int(math.floor(pz / 0.012))
                    if hash3(ci, 81, cj) < 0.0022:
                        ex = px / 0.012 - ci - 0.5; ez = pz / 0.012 - cj - 0.5
                        if ex * ex + ez * ez < 0.02 + 0.12 * hash3(ci, 83, cj): sc += 0.6
                    sc = min(sc, 0.85)
                    lr = lr * (1 - sc) + sc * 0.36 * er / math.pi; lg = lg * (1 - sc) + sc * 0.34 * eg / math.pi
                    lb = lb * (1 - sc) + sc * 0.28 * eb / math.pi
                    cr += lr; cg += lg; cb += lb
                    continue
                gx, gy, gz = normal(px, py, pz, t)
                fp = t * pa / max(-(gx * dx + gy * dy + gz * dz), 0.05)
                ar, ag, ab, f0r, f0g, f0b, ro, nx, ny, nz = material(px, py, pz, gx, gy, gz, m, fp)
                er, eg, eb, sr, sg, sb = lamps_at(px + gx * 0.003, py + gy * 0.003, pz + gz * 0.003, nx, ny, nz, dx, dy, dz, ro, q, N, pr)
                ok, ir, ig, ib = bounce_at(BP, BN, BE, fx / step, fy / step, px, py, pz, gx, gy, gz, t)
                if not ok:
                    ir, ig, ib = grid_irr(G, GV, px, py, pz, gx, gy, gz)
                occ = 0.0; sca = 1.0
                for k in range(4):
                    hh = 0.02 + 0.09 * k
                    occ += (hh - scene(px + gx * hh, py + gy * hh, pz + gz * hh)[0]) * sca; sca *= 0.7
                ao = min(max(1.0 - 1.6 * occ, 0.0), 1.0)
                er += ir + AMB[0] * ao; eg += ig + AMB[1] * ao; eb += ib + AMB[2] * ao
                lr = ar * er / math.pi; lg = ag * eg / math.pi; lb = ab * eb / math.pi
                # gloss
                if f0r > 0.0 and ro < 0.6:
                    vn = dx * nx + dy * ny + dz * nz
                    c1 = 1.0 - min(max(-vn, 0.0), 1.0)
                    fk = c1 ** 5 * max(1.0 - 1.5 * ro, 0.0)
                    Fr = f0r + (1 - f0r) * fk; Fg = f0g + (1 - f0g) * fk; Fb = f0b + (1 - f0b) * fk
                    qr = 0.0; qg = 0.0; qb = 0.0
                    if ro < 0.4:
                        mx0 = dx - 2 * vn * nx; my0 = dy - 2 * vn * ny; mz0 = dz - 2 * vn * nz
                        if abs(my0) < 0.9: tx = -mz0; ty = 0.0; tz = mx0            # two directions across the mirror direction
                        else: tx = 1.0; ty = 0.0; tz = 0.0
                        tl = math.sqrt(tx * tx + ty * ty + tz * tz); tx /= tl; ty /= tl; tz /= tl
                        bx = my0 * tz - mz0 * ty; by = mz0 * tx - mx0 * tz; bz = mx0 * ty - my0 * tx
                        ox = px + gx * 0.003; oy = py + gy * 0.003; oz = pz + gz * 0.003
                        NR = 3 if ro > 0.1 else 2
                        for k in range(NR):
                            s3 = ((s * 11 + sh * 3 + 5) % N) * NR + k
                            u = (s3 + hash3(ix + k * 131, iy + s, seed + 17)) / (N * NR)
                            rr = ro * 0.55 * math.sqrt(-math.log(1.0 - 0.96 * u)); th = 2.39996323 * s3 + pr2
                            rx = mx0 + rr * (math.cos(th) * tx + math.sin(th) * bx); ry = my0 + rr * (math.cos(th) * ty + math.sin(th) * by)
                            rz = mz0 + rr * (math.cos(th) * tz + math.sin(th) * bz)
                            rl = math.sqrt(rx * rx + ry * ry + rz * rz); rx /= rl; ry /= rl; rz /= rl
                            gn = rx * gx + ry * gy + rz * gz
                            if gn < 0.01:
                                rx += (0.01 - gn) * gx; ry += (0.01 - gn) * gy; rz += (0.01 - gn) * gz
                                rl = math.sqrt(rx * rx + ry * ry + rz * rz); rx /= rl; ry /= rl; rz /= rl
                            ur, ug, ub = plain(ox, oy, oz, rx, ry, rz, t, G, GV, AMB, ro < 0.03, (q + k * 5) % N, N, pr2, pa)
                            qr += ur; qg += ug; qb += ub
                        qr /= NR; qg /= NR; qb /= NR
                    else:
                        qr = lr; qg = lg; qb = lb                                    # too rough to mirror anything: only the lamps' sheen
                    if ro >= 0.03:
                        qr += sr; qg += sg; qb += sb
                    lr = lr * (1 - Fr) + Fr * qr; lg = lg * (1 - Fg) + Fg * qg; lb = lb * (1 - Fb) + Fb * qb
                cr += lr; cg += lg; cb += lb
            img[iy, ix, 0] = cr / N; img[iy, ix, 1] = cg / N; img[iy, ix, 2] = cb / N
    return img

def develop(lin, scale, rng, exposure):
    """lens, then a restrained curve, then grain."""
    H, W = lin.shape[:2]
    lin = lin * exposure
    # the lens: a little softness, a little veiling light, falloff about the axis (which is at eye height, not mid-frame)
    soft = np.stack([gaussian_filter(lin[..., c], 0.55 * scale) for c in range(3)], -1)
    veil = np.stack([gaussian_filter(np.minimum(lin[..., c], 40.0), 0.045 * W) for c in range(3)], -1)
    halo = np.stack([gaussian_filter(np.minimum(lin[..., c], 40.0), 3.5 * scale) for c in range(3)], -1)
    glow = np.stack([gaussian_filter(np.minimum(lin[..., c], 40.0), 14.0 * scale) for c in range(3)], -1)
    haze = np.stack([gaussian_filter(np.minimum(lin[..., c], 40.0), 48.0 * scale) for c in range(3)], -1)   # sea air round the lamp
    lin = soft * 0.951 + halo * 0.022 + glow * 0.012 + haze * 0.006 + veil * 0.009
    u = ((np.arange(W) + 0.5) / W * 2 - 1) * TU
    v = VTOP - (np.arange(H) + 0.5) / H * (VTOP - VBOT)
    c4 = 1.0 / (1.0 + u[None, :] ** 2 + v[:, None] ** 2) ** 2
    lin = lin * (1.0 - 0.30 * (1.0 - c4))[..., None]
    # the curve
    wp = 9.0
    f = lambda x: x * (1.0 + x / wp ** 2) / (1.0 + x)
    L = (0.2126 * lin[..., 0] + 0.7152 * lin[..., 1] + 0.0722 * lin[..., 2])[..., None]
    y = 0.6 * lin * (f(L) / np.maximum(L, 1e-9)) + 0.4 * f(lin)             # mostly by brightness, so lit paint keeps its colour
    y = np.clip(y, 0.0, 1.0)
    y = np.where(y <= 0.0031308, 12.92 * y, 1.055 * np.power(np.maximum(y, 1e-9), 1 / 2.4) - 0.055)
    y = y + 0.20 * (y * y * (3 - 2 * y) - y)                                  # a slight S
    # grain: fine, strongest in the middle tones
    g = gaussian_filter(rng.standard_normal((H, W)), 0.55 * max(scale, 0.5)); g /= g.std()
    gc = np.stack([gaussian_filter(rng.standard_normal((H, W)), 0.55 * max(scale, 0.5)) for _ in range(3)], -1); gc /= gc.std()
    lum = y.mean(-1, keepdims=True)
    amp = 0.003 + 0.008 * np.sqrt(np.clip(4 * lum * (1 - lum), 0, 1))
    y = y + amp * (0.8 * g[..., None] + 0.35 * gc)
    return np.clip(y, 0, 1)


def stamp(img, rng):
    """H.M.Q. / 25:00, violet stamp-pad ink, a little dry, a few degrees off level, lower left."""
    H, W = img.shape[:2]
    fs = max(int(round(min(H, W) * 0.027)), 9)
    font = ImageFont.truetype('/usr/share/fonts/opentype/urw-base35/NimbusMonoPS-Bold.otf', fs)
    cw, chh = int(fs * 4.7), int(fs * 2.95)
    im = Image.new('L', (cw, chh), 0); d = ImageDraw.Draw(im)
    bw = max(int(round(fs * 0.085)), 1)
    d.rounded_rectangle([bw, bw, cw - bw - 1, chh - bw - 1], radius=int(fs * 0.22), outline=255, width=bw)
    for k, s in enumerate(('H.M.Q.', '25:00')):
        l = d.textlength(s, font=font)
        d.text(((cw - l) / 2, fs * (0.34 + 1.12 * k)), s, fill=255, font=font)
    im = im.rotate(4.2, resample=Image.BICUBIC, expand=True)
    m = np.asarray(im, np.float64) / 255.0
    h, w = m.shape
    # rubber: ink gathers at the edge of each letter, starves in the middle of it
    inner = gaussian_filter(m, fs * 0.055)
    dens = m * (0.62 + 0.75 * (1.0 - inner))
    # uneven pressure across the stamp, one corner hardly touching
    press = gaussian_filter(rng.standard_normal((h, w)), fs * 0.9); press = press / (np.abs(press).max() + 1e-9)
    yy, xx = np.mgrid[0:h, 0:w]
    corner = np.clip(1.25 - 1.05 * np.clip((xx / w - 0.45) / 0.55, 0, 1) * np.clip((0.75 - yy / h) / 0.75 + 0.35, 0, 1), 0.12, 1)
    dens *= np.clip(0.80 + 0.45 * press, 0.2, 1.1) * corner
    # the pad was a little dry: small misses where the rubber met no ink
    dry = gaussian_filter(rng.standard_normal((h, w)), fs * 0.028); dry /= dry.std()
    miss = np.clip((dry - 0.75 + 1.2 * (1 - corner)) / 0.6, 0, 1)
    dens *= 1.0 - 0.75 * miss
    dens = gaussian_filter(dens, fs * 0.022)
    dens = np.clip(dens * 1.15, 0, 1) * 0.86
    ink = np.array([0.36, 0.20, 0.60])
    x0 = int(W * 0.050); y0 = H - int(W * 0.050) - h
    reg = img[y0:y0 + h, x0:x0 + w]
    out = reg * (1.0 - dens[..., None] * (1.0 - ink))                         # ink absorbs
    out = out + dens[..., None] * 0.46 * (ink * 0.9 - out) * np.clip((0.55 - reg.mean(-1, keepdims=True)) / 0.3, 0, 1)   # and sits on the surface in the darks
    img[y0:y0 + h, x0:x0 + w] = out
    return img
