#!/usr/bin/env python
"""
Winter 1971, a whale went under me
==================================
Go Sun-deok (seal: 숨).  Ink wash on hanji, laid by program.  Her largest sheet.

    python winter_1971_a_whale_went_under_me.py [scale] [out.png]

scale 1.0 -> 2400 x 3800 px (default); 0.35 for a quick study.

Seed: the tide table for Hado-ri on the day of the dive (see TIDE below).

How the sheet is made (the same engine as the other sheets, copied here whole)
------------------------------------------------------------------------------
* Paper: a hanji sheet is built first - cloudy thickness, tens of thousands of
  long mulberry fibres laid as curved threads (they give the tooth, the pale
  and dark threads, and the direction water prefers to travel).
* Ink is optical density, never colour: colour = paper * exp(-density * k(rgb)),
  with k leaning blue-grey for thin ink and warm for thick (pine soot).
* Wet: a half-resolution grid keeps water, mobile pigment and settled pigment
  apart.  Water creeps by capillary flow (only out of cells wet enough to give
  it), pigment rides the water and diffuses only between wet cells, settles
  faster into the hollows of the paper, and whatever the flow carried to the
  edge of the wet is left there as the drying line.  Fresh water arriving in a
  damp wash lifts pigment again and pushes it out: a back-run.
* Reserves: the scars are marked before any wash; less ink is laid on them and
  the water is slowed there, so they stay pale.  Nothing is ever painted pale.
* Dry (galpil): one gesture along a spline with pressure and ink load varying
  along it.  Each bristle carries its own ink; as the load falls, only the
  best-loaded bristles on the highest fibres still mark the paper.
* Light is paper.  There is no white anywhere in this file.

This sheet
----------
The one time she lets the wash be larger than the paper it leaves.  The whale
is a single flooded wash, begun with a loaded brush at the bottom of the sheet
and dragged up and out to the tail until the brush has nothing left; dark ink
is dropped into it along the spine while it is wet; clean water flicked on as
it goes damp makes the pale blooms a gray whale carries on its back.  The top
third of the sheet is air and is not touched, except by her, the rope and the
float.
"""

# ----------------------------------------------------------------- helpers
def smooth_noise(rng, shape, sigma):
    n = ndi.gaussian_filter(rng.standard_normal(shape).astype(np.float32), sigma, mode="wrap")
    return n / (n.std() + 1e-9)


def rank01(a):
    """map to a roughly uniform 0..1 distribution (via the normal cdf of the z-score)"""
    z = (a - a.mean()) / (a.std() + 1e-9)
    return (0.5 * (1.0 + np.vectorize(math.erf, otypes=[np.float32])(z / 1.41421356))).astype(np.float32) \
        if a.size < 4096 else _erf01(z.astype(np.float32))


@njit(parallel=True, cache=True)
def _erf01(z):
    out = np.empty_like(z)
    f = z.ravel()
    o = out.ravel()
    for i in prange(f.size):
        o[i] = 0.5 * (1.0 + math.erf(f[i] / 1.41421356))
    return out


def crom(pts, spacing):
    """Catmull-Rom through pts, resampled at even spacing. returns xy (n,2), length"""
    pts = np.asarray(pts, dtype=np.float64)
    P = np.vstack([2 * pts[0] - pts[1], pts, 2 * pts[-1] - pts[-2]])
    out = []
    for i in range(len(pts) - 1):
        p0, p1, p2, p3 = P[i:i + 4]
        n = max(6, int(np.linalg.norm(p2 - p1) / max(spacing, 0.25) * 1.5))
        t = np.linspace(0, 1, n, endpoint=False)[:, None]
        out.append(0.5 * ((2 * p1) + (-p0 + p2) * t + (2 * p0 - 5 * p1 + 4 * p2 - p3) * t * t
                          + (-p0 + 3 * p1 - 3 * p2 + p3) * t ** 3))
    out.append(pts[-1:])
    xy = np.vstack(out)
    d = np.r_[0, np.cumsum(np.hypot(*np.diff(xy, axis=0).T))]
    n = max(8, int(d[-1] / spacing))
    s = np.linspace(0, d[-1], n)
    return np.c_[np.interp(s, d, xy[:, 0]), np.interp(s, d, xy[:, 1])], d[-1]


def profile(spec, n, smooth=0.02):
    """spec: list of (t, value) -> smooth array of n samples"""
    spec = np.asarray(spec, dtype=np.float64)
    v = np.interp(np.linspace(0, 1, n), spec[:, 0], spec[:, 1])
    if smooth > 0 and n > 8:
        v = ndi.gaussian_filter1d(v, max(1.0, smooth * n), mode="nearest")
    return v


# ------------------------------------------------------------------- paper
@njit(cache=True, fastmath=True)
def _fibres(F, TX, TY, n, lmin, lmax, wmin, wmax, alo, ahi, wander, seed):
    np.random.seed(seed)
    h, w = F.shape
    th_, tw_ = TX.shape
    step = 0.7
    for i in range(n):
        x = np.random.random() * (w + 200) - 100
        y = np.random.random() * (h + 200) - 100
        th = np.random.random() * 6.2831853
        if np.random.random() < 0.35:            # the vat's rocking lays a share of them one way
            th = 0.25 * np.random.standard_normal()
        L = lmin * (lmax / lmin) ** np.random.random()
        wd = wmin + (wmax - wmin) * np.random.random() ** 2
        a = alo + (ahi - alo) * np.random.random()
        curv = (np.random.random() - 0.5) * wander
        steps = int(L / step)
        sg = max(0.45, wd * 0.5)
        R = int(sg * 2.5) + 1
        norm = a * step / (sg * 2.5)
        for k in range(steps):
            t = k / steps
            tap = min(1.0, 5 * t, 5 * (1 - t))
            th += curv * step + (np.random.random() - 0.5) * wander * 6
            c = math.cos(th)
            s = math.sin(th)
            x += c * step
            y += s * step
            ix = int(x)
            iy = int(y)
            if ix < -R or iy < -R or ix >= w + R or iy >= h + R:
                continue
            for yy in range(max(0, iy - R), min(h, iy + R + 1)):
                for xx in range(max(0, ix - R), min(w, ix + R + 1)):
                    d2 = (xx - x) ** 2 + (yy - y) ** 2
                    F[yy, xx] += norm * tap * math.exp(-d2 / (2 * sg * sg))
            sx = ix // 2
            sy = iy // 2
            if 0 <= sx < tw_ and 0 <= sy < th_:
                TX[sy, sx] += c * c * wd
                TY[sy, sx] += s * s * wd


# ----------------------------------------------------------------- wet sim
@njit(cache=True, fastmath=True, inline="always")
def _mob(w, wmin, wr):
    m = (w - wmin) / wr
    if m < 0.0:
        return 0.0
    if m > 1.0:
        return 1.0
    return m


@njit(cache=True, fastmath=True, inline="always")
def _edge(wi, pi, wj, pj, k, wmin, wr, carry, dp):
    """flux of water and pigment from i to j across one edge"""
    dw = wi - wj
    mi = _mob(wi, wmin, wr)
    mj = _mob(wj, wmin, wr)
    if dw > 0:
        f = k * dw * mi
        fp = carry * f * pi / (wi + 1e-4)
    else:
        f = k * dw * mj
        fp = carry * f * pj / (wj + 1e-4)
    g = mi if mi < mj else mj
    fp += dp * k * g * (pi - pj)
    return f, fp


@njit(parallel=True, cache=True, fastmath=True)
def _flow(W, P, D, KX, KY, HOL, n, dt, wmin, wr, evap, carry, dp, fix0, fix1, gran, lift):
    h, w = W.shape
    W2 = np.empty_like(W)
    P2 = np.empty_like(P)
    for it in range(n):
        for y in prange(h):
            for x in range(w):
                wi = W[y, x]
                pi = P[y, x]
                if wi <= 0.0 and pi <= 0.0:
                    dry = True
                    if x > 0 and W[y, x - 1] > wmin:
                        dry = False
                    if x < w - 1 and W[y, x + 1] > wmin:
                        dry = False
                    if y > 0 and W[y - 1, x] > wmin:
                        dry = False
                    if y < h - 1 and W[y + 1, x] > wmin:
                        dry = False
                    if dry:
                        W2[y, x] = 0.0
                        P2[y, x] = 0.0
                        continue
                fw = 0.0
                fp = 0.0
                if x > 0:
                    a, b = _edge(wi, pi, W[y, x - 1], P[y, x - 1], KX[y, x - 1], wmin, wr, carry, dp)
                    fw += a
                    fp += b
                if x < w - 1:
                    a, b = _edge(wi, pi, W[y, x + 1], P[y, x + 1], KX[y, x], wmin, wr, carry, dp)
                    fw += a
                    fp += b
                if y > 0:
                    a, b = _edge(wi, pi, W[y - 1, x], P[y - 1, x], KY[y - 1, x], wmin, wr, carry, dp)
                    fw += a
                    fp += b
                if y < h - 1:
                    a, b = _edge(wi, pi, W[y + 1, x], P[y + 1, x], KY[y, x], wmin, wr, carry, dp)
                    fw += a
                    fp += b
                wn = wi - dt * fw
                pn = pi - dt * fp
                if pn < 0.0:
                    pn = 0.0
                # fresh water arriving lifts settled pigment of this sitting (back-runs)
                inflow = wn - wi
                if inflow > 0.0 and lift > 0.0:
                    lf = lift * inflow
                    if lf > 0.5:
                        lf = 0.5
                    lf *= D[y, x]
                    D[y, x] -= lf
                    pn += lf
                wn -= evap
                if wn < 0.0:
                    wn = 0.0
                # settling: slow while flooded, fast as it goes damp, more into hollows
                m = _mob(wn, wmin, wr)
                rate = (fix0 + fix1 * (1.0 - m)) * (1.0 + gran * (HOL[y, x] - 0.5))
                if wn <= 1e-4:
                    rate = 1.0
                if rate > 1.0:
                    rate = 1.0
                if rate < 0.0:
                    rate = 0.0
                dep = pn * rate
                D[y, x] += dep
                P2[y, x] = pn - dep
                W2[y, x] = wn
        for y in prange(h):
            for x in range(w):
                W[y, x] = W2[y, x]
                P[y, x] = P2[y, x]


# ------------------------------------------------------------------- brush
@njit(cache=True, fastmath=True)
def _stroke(px, py, rad, load, tone, CAP, wob, H, Dd, Pa, Wa, conc, water, wet, soft, tooth, body):
    n = px.size
    h, w = H.shape
    nb, ns = CAP.shape
    rmax = rad.max() + 3.0
    x0 = max(0, int(px.min() - rmax))
    x1 = min(w, int(px.max() + rmax) + 2)
    y0 = max(0, int(py.min() - rmax))
    y1 = min(h, int(py.max() + rmax) + 2)
    if x1 <= x0 or y1 <= y0:
        return
    bw = x1 - x0
    bh = y1 - y0
    best = np.full((bh, bw), 1e18, dtype=np.float32)
    lat = np.zeros((bh, bw), dtype=np.float32)
    idx = np.full((bh, bw), -1, dtype=np.int32)
    for i in range(n):
        i0 = i - 1 if i > 0 else 0
        i1 = i + 1 if i < n - 1 else n - 1
        tx = px[i1] - px[i0]
        ty = py[i1] - py[i0]
        tl = math.sqrt(tx * tx + ty * ty) + 1e-9
        tx /= tl
        ty /= tl
        R = rmax          # search by plain distance: the nearest point of the gesture owns the pixel
        R2 = R * R
        for yy in range(max(y0, int(py[i] - R)), min(y1, int(py[i] + R) + 2)):
            for xx in range(max(x0, int(px[i] - R)), min(x1, int(px[i] + R) + 2)):
                dx = xx - px[i]
                dy = yy - py[i]
                d2 = dx * dx + dy * dy
                if d2 < R2 and d2 < best[yy - y0, xx - x0]:
                    best[yy - y0, xx - x0] = d2
                    idx[yy - y0, xx - x0] = i
                    la = dx * ty - dy * tx
                    if i == 0 or i == n - 1:
                        al = dx * tx + dy * ty
                        if (i == 0 and al < 0) or (i == n - 1 and al > 0):
                            la = 1e9          # the brush is lifted: nothing beyond the end of the gesture
                    lat[yy - y0, xx - x0] = la
    for yy in range(bh):
        for xx in range(bw):
            i = idx[yy, xx]
            if i < 0:
                continue
            r = rad[i]
            la = lat[yy, xx]
            d = abs(la)
            e = (r - d) / soft + 0.5
            if e <= 0.0:
                continue
            if e > 1.0:
                e = 1.0
            L = load[i]
            u = la / (r + 1e-6)
            if u < -1.0:
                u = -1.0
            if u > 1.0:
                u = 1.0
            fb = (u * 0.5 + 0.5) * (nb - 1) + wob[i]
            if fb < 0:
                fb = 0.0
            if fb > nb - 1.001:
                fb = nb - 1.001
            b0 = int(fb)
            fr = fb - b0
            si = int(i * ns / n)
            cap = CAP[b0, si] * (1 - fr) + CAP[b0 + 1, si] * fr
            Hp = H[yy + y0, xx + x0]
            thr = 1.0 - L
            v = 0.62 * cap + tooth * Hp + 0.18 - thr * (0.80 + tooth)
            c = v / 0.16
            if c <= 0.0:
                continue
            if c > 1.0:
                c = 1.0
            c = c * c * (3 - 2 * c)
            amt = conc * tone[i] * c * e * (0.74 + 0.26 * cap) * (body + (1.0 - body) * L)
            wf = (L - 0.30) / 0.45
            if wf < 0.0:
                wf = 0.0
            if wf > 1.0:
                wf = 1.0
            wf *= wet
            Dd[yy + y0, xx + x0] += amt * (1.0 - wf)
            Pa[yy + y0, xx + x0] += amt * wf
            Wa[yy + y0, xx + x0] += water * wf * e


# ------------------------------------------------------------------- sheet
class Sheet:
    def __init__(self, bw, bh, S, seed):
        self.S = S
        self.W = int(round(bw * S / 2)) * 2
        self.H = int(round(bh * S / 2)) * 2
        self.sw, self.sh = self.W // 2, self.H // 2
        self.S2 = S * S
        self.rng = np.random.default_rng(seed)
        self.seed = seed
        self._paper()
        z = lambda: np.zeros((self.sh, self.sw), np.float32)
        self.Ww, self.Pp, self.Dc, self.Dfix = z(), z(), z(), z()
        f = lambda: np.zeros((self.H, self.W), np.float32)
        self.Dd, self.Pa, self.Wa = f(), f(), f()
        self.seals = []
        self.Rfull = np.ones((self.H, self.W), np.float32)      # reserves: 0 where paper is kept
        self.Rsim = np.ones((self.sh, self.sw), np.float32)

    # .................................................. the hanji
    def _paper(self):
        S, W, H, rng = self.S, self.W, self.H, self.rng
        F_long = np.zeros((H, W), np.float32)
        F_fine = np.zeros((H, W), np.float32)
        F_dark = np.zeros((H, W), np.float32)
        TX = np.zeros((self.sh, self.sw), np.float32)
        TY = np.zeros((self.sh, self.sw), np.float32)
        dump = np.zeros((2, 2), np.float32)
        area = W * H / (1700 * 3000.0) / (S * S)
        # long mulberry fibres: the visible threads
        _fibres(F_long, TX, TY, int(5200 * area), 260 * S, 1500 * S, 0.7 * S + 0.35, 2.3 * S + 0.5,
                0.35, 1.0, 0.0030 / S, self.seed % 100000 + 1)
        # the felt of fine fibre that makes the tooth
        _fibres(F_fine, TX, TY, int(150000 * area), 30 * S, 190 * S, 0.5, 0.75 * S + 0.3,
                0.3, 1.0, 0.012 / S, self.seed % 100000 + 2)
        # a few darker strands (inner bark left in the pulp)
        _fibres(F_dark, dump, dump, int(260 * area), 60 * S, 420 * S, 0.6 * S + 0.3, 1.5 * S + 0.4,
                0.3, 1.0, 0.004 / S, self.seed % 100000 + 3)
        cloud = (0.45 * smooth_noise(rng, (H, W), 5 * S) + 0.55 * smooth_noise(rng, (H, W), 16 * S)
                 + 0.65 * smooth_noise(rng, (H, W), 55 * S) + 0.5 * smooth_noise(rng, (H, W), 190 * S))
        cloud /= cloud.std()
        grain = smooth_noise(rng, (H, W), 0.7)
        fl = F_long / (F_long.std() + 1e-9)
        ff = F_fine / (F_fine.std() + 1e-9)
        tooth = 0.6 * ff + 0.22 * np.minimum(fl, 4.0) + 0.5 * grain + 0.25 * cloud
        self.Ht = _erf01(((tooth - tooth.mean()) / tooth.std()).astype(np.float32))
        self.cloud = cloud.astype(np.float32)
        # colour of the bare sheet
        fl_s = np.minimum(ndi.gaussian_filter(F_long, 0.5), 2.2)
        fd = np.minimum(ndi.gaussian_filter(F_dark, 0.5), 1.6)
        lum = (1.0 + 0.011 * cloud + 0.011 * (fl_s - fl_s.mean()) + 0.004 * (ff - ff.mean()).clip(-2, 3)
               - 0.05 * fd)
        # raking light on the tooth
        hs = ndi.gaussian_filter(self.Ht, 0.6)
        lum += 0.022 * (hs - np.roll(np.roll(hs, 1, 0), 1, 1))
        base = np.array([0.946, 0.928, 0.884], np.float32)
        paper = base[None, None, :] * lum[..., None].astype(np.float32)
        paper[..., 2] *= (1.0 + 0.006 * cloud - 0.02 * fd)      # thicker = warmer
        paper[..., 0] *= (1.0 + 0.004 * cloud)
        self.paper = np.clip(paper, 0, 1).astype(np.float32)
        self.Flong = fl_s.astype(np.float32)
        # what the water sees (half resolution)
        ds = lambda a: a.reshape(self.sh, 2, self.sw, 2).mean(axis=(1, 3))
        TXs = ndi.gaussian_filter(TX, 2.5 * max(S, 0.5))
        TYs = ndi.gaussian_filter(TY, 2.5 * max(S, 0.5))
        ani = (TXs - TYs) / (TXs + TYs + 1e-6)
        perm = np.exp(0.38 * ds(cloud) + 0.30 * smooth_noise(rng, (self.sh, self.sw), 1.2)
                      + 0.12 * ds(np.minimum(fl, 3.0) - fl.mean()))
        self.KX = np.clip(0.62 * perm * (1 + 0.55 * ani), 0.02, 1.15).astype(np.float32)
        self.KY = np.clip(0.62 * perm * (1 - 0.55 * ani), 0.02, 1.15).astype(np.float32)
        self.HOL = (-ds(0.55 * ff + 0.45 * grain + 0.12 * np.minimum(fl, 3.0))).astype(np.float32)
        self.HOL = ndi.gaussian_filter(self.HOL, 0.8)
        self.HOL = np.clip(0.5 + (self.HOL - self.HOL.mean()) / (4.0 * self.HOL.std()), 0, 1).astype(np.float32)

    # .................................................. laying ink
    def stroke(self, pts, rad, load, conc, water=0.0, wet=0.0, nb=48, soft=1.2, tooth=0.6,
               body=0.7, wobble=1.2, seed=0, streak=1.0, tone=None):
        S = self.S
        pts = np.asarray(pts, np.float64) * S
        rmax = max(r for _, r in rad) * S
        spacing = max(0.5, min(rmax / 14.0, 3.0))
        xy, length = crom(pts, spacing)
        n = len(xy)
        r = np.maximum(profile(rad, n) * S, 0.0)
        L = np.clip(profile(load, n), 0, 1)
        tn = np.ones(n) if tone is None else profile(tone, n)
        rg = np.random.default_rng(self.seed + 7919 * (seed + 1))
        ns = 256
        pad = 8                                      # so the outermost hairs are no different from the rest
        cap = ndi.gaussian_filter(rg.standard_normal((nb + 1 + 2 * pad, ns)).astype(np.float32),
                                  (0.6, 30.0 / streak), mode="wrap")
        clump = ndi.gaussian_filter(rg.standard_normal((nb + 1 + 2 * pad, ns)).astype(np.float32), (2.5, 60.0),
                                    mode="wrap")
        cap = (cap / cap.std() + 0.6 * clump / clump.std())[pad:-pad]
        CAP = _erf01(((cap - cap.mean()) / cap.std()).astype(np.float32))
        wob = ndi.gaussian_filter1d(rg.standard_normal(n), max(2.0, 40.0 * S / spacing))
        wob = wobble * wob / (np.abs(wob).max() + 1e-9)
        _stroke(xy[:, 0].astype(np.float32), xy[:, 1].astype(np.float32), r.astype(np.float32),
                L.astype(np.float32), tn.astype(np.float32), CAP, wob.astype(np.float32), self.Ht, self.Dd, self.Pa, self.Wa,
                conc, water, wet, max(0.8, soft * S), tooth, body)

    def reserve(self, x, y, rx, ry=None, ang=0.0, seed=0, edge=1.5):
        """paper to be kept (a bubble, foam): planned before any wash. Nothing wet is laid there and
        the water is held off it, so the wash runs round and dries against its edge."""
        S = self.S
        ry = rx if ry is None else ry
        R = int(max(rx, ry) * S * 1.4) + 6
        x0, x1 = max(0, int(x * S) - R), min(self.W, int(x * S) + R)
        y0, y1 = max(0, int(y * S) - R), min(self.H, int(y * S) + R)
        x0 -= x0 % 2
        y0 -= y0 % 2
        x1 -= x1 % 2
        y1 -= y1 % 2
        yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
        dx, dy = xx - x * S, yy - y * S
        ca, sa = math.cos(ang), math.sin(ang)
        u, v = (dx * ca + dy * sa) / (rx * S), (-dx * sa + dy * ca) / (ry * S)
        th = np.arctan2(v, u)
        rg = np.random.default_rng(self.seed + 7717 * (seed + 1))
        k = np.zeros_like(th)
        for m in range(2, 6):
            k += 0.05 * rg.standard_normal() / (m - 1) * np.cos(m * th + rg.uniform(0, 6.28))
        d = (np.hypot(u, v) - (1 + k)) * min(rx, ry) * S          # signed distance in px, roughly
        keep = np.clip(d / max(edge * S, 0.8) + 0.5, 0, 1).astype(np.float32)
        self.Rfull[y0:y1, x0:x1] *= keep
        ks = keep.reshape((y1 - y0) // 2, 2, (x1 - x0) // 2, 2).min(axis=(1, 3))
        sy, sx = slice(y0 // 2, y1 // 2), slice(x0 // 2, x1 // 2)
        self.Rsim[sy, sx] *= ks
        self.KX[sy, sx] *= np.minimum(ks, np.roll(ks, -1, axis=1))
        self.KY[sy, sx] *= np.minimum(ks, np.roll(ks, -1, axis=0))

    def drop(self, x, y, r, pig, water, ragged=0.25, seed=0, squash=1.0, ang=0.0):
        """ink (or clean water) touched into the sheet: goes straight to the wet layer"""
        S = self.S
        cx, cy, rr = x * S / 2, y * S / 2, r * S / 2
        R = int(rr * 1.6) + 3
        x0, x1 = max(0, int(cx) - R), min(self.sw, int(cx) + R)
        y0, y1 = max(0, int(cy) - R), min(self.sh, int(cy) + R)
        if x1 <= x0 or y1 <= y0:
            return
        yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
        dx, dy = xx - cx, yy - cy
        ca, sa = math.cos(ang), math.sin(ang)
        u = dx * ca + dy * sa
        v = (-dx * sa + dy * ca) / squash
        th = np.arctan2(v, u)
        rg = np.random.default_rng(self.seed + 104729 * (seed + 1))
        k = np.zeros_like(th)
        for m in range(2, 9):
            k += rg.standard_normal() / m * np.cos(m * th + rg.uniform(0, 6.28))
        d = np.hypot(u, v) / (rr * (1 + ragged * k))
        a = np.clip((1.0 - d) / 0.18, 0, 1)
        a = a * a * (3 - 2 * a)
        a = a * self.Rsim[y0:y1, x0:x1]
        self.Pp[y0:y1, x0:x1] += pig * a
        self.Ww[y0:y1, x0:x1] = np.minimum(self.Ww[y0:y1, x0:x1] + water * a, 1.6)

    def patch(self, poly, pig, water, seed=0, rough=0.3, fade=None):
        """ink laid with the side of the brush as one irregular mass, straight into the wet layer.
        fade=(y0, y1, floor): the mass thins out downward between y0 and y1"""
        k = self.S / 2
        im = Image.new("L", (self.sw, self.sh), 0)
        ImageDraw.Draw(im).polygon([(x * k, y * k) for x, y in poly], fill=255)
        rg = np.random.default_rng(self.seed + 15485863 * (seed + 1))
        m = ndi.gaussian_filter(np.asarray(im, np.float32) / 255.0, 5.0 * self.S)
        nz = 0.6 * smooth_noise(rg, m.shape, 3.0 * self.S + 0.5) + 0.8 * smooth_noise(rg, m.shape, 14.0 * self.S)
        a = np.clip((m + rough * nz * np.sqrt(np.clip(m * (1 - m), 0, 1)) - 0.5) / 0.22 + 0.5, 0, 1)
        a = a * a * (3 - 2 * a)
        body = np.clip(1.0 + 0.16 * smooth_noise(rg, m.shape, 30.0 * self.S), 0.5, 1.5)
        if fade is not None:
            yy = np.arange(self.sh, dtype=np.float32)[:, None] / k
            t = np.clip((yy - fade[0]) / (fade[1] - fade[0]) + 0.12 * smooth_noise(rg, m.shape, 40.0 * self.S), 0, 1)
            f_ = 1.0 - (1.0 - fade[2]) * t * t * (3 - 2 * t)
            body = body * f_
        a = a * self.Rsim
        self.Pp += (pig * a * body).astype(np.float32)
        self.Ww = np.minimum(self.Ww + water * a, 1.6).astype(np.float32)

    def grid(self):
        """coordinates of the wet layer's cells, in the sheet's own units (scale 1 pixels)"""
        k = 2.0 / self.S
        yy, xx = np.mgrid[0:self.sh, 0:self.sw].astype(np.float32)
        return (xx + 0.5) * k, (yy + 0.5) * k

    def lay(self, pig, water):
        """a broad wash floated on with the big brush: pigment and water given over the whole sheet"""
        self.Pp += (pig * self.Rsim).astype(np.float32)
        self.Ww = np.minimum(self.Ww + water * self.Rsim, 1.6).astype(np.float32)

    def flush(self):
        ds = lambda a: a.reshape(self.sh, 2, self.sw, 2).mean(axis=(1, 3))
        self.Pp += ds(self.Pa * self.Rfull)
        self.Ww = np.minimum(self.Ww + ds(self.Wa * self.Rfull), 1.6).astype(np.float32)
        self.Pa[:] = 0
        self.Wa[:] = 0

    def flow(self, t, evap=0.0006, carry=0.8, dp=0.35, fix0=0.0004, fix1=0.02, gran=0.4, lift=0.6,
             wmin=0.22, wr=0.25):
        """let the sheet sit for t (time is in steps at scale 1; scaled so pictures agree at any scale)"""
        self.flush()
        n = max(1, int(round(t * self.S2)))
        s2 = self.S2
        _flow(self.Ww, self.Pp, self.Dc, self.KX, self.KY, self.HOL, n, 0.2, wmin, wr,
              evap / s2, carry, dp, min(1.0, fix0 / s2), min(1.0, fix1 / s2), gran, lift)

    def dry(self, **kw):
        """wait until the sheet is dry; ink of this sitting can no longer be lifted"""
        self.flush()
        for _ in range(60):
            if float(self.Ww.max()) <= 1e-4:
                break
            self.flow(400, **kw)
        self.Dfix += self.Dc + self.Pp
        self.Dc[:] = 0
        self.Pp[:] = 0
        self.Ww[:] = 0

    # .................................................. the seal
    def seal(self, cx, cy, size, rot=0.0, seed=0, firm=0.0):
        self.seals.append((cx, cy, size, rot, seed, firm))

    def _press_seal(self, col, cx, cy, size, rot, seed, firm=0.0):
        """vermilion square seal, white-on-red: 숨 built from its three jamo"""
        S = self.S
        s = size * S
        R = int(s * 0.8) + 4
        x0, y0 = int(cx * S) - R, int(cy * S) - R
        x1, y1 = x0 + 2 * R, y0 + 2 * R
        if x0 < 0 or y0 < 0 or x1 > self.W or y1 > self.H:
            raise ValueError("seal off the sheet")
        rg = np.random.default_rng(self.seed + 31337 * (seed + 1))
        yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
        ca, sa = math.cos(math.radians(rot)), math.sin(math.radians(rot))
        dx, dy = (xx - cx * S) / s, (yy - cy * S) / s
        u = dx * ca + dy * sa
        v = -dx * sa + dy * ca
        # the carved edge is never perfectly straight
        wig = 0.007 * smooth_noise(rg, u.shape, 3.0 * S + 0.5) + 0.004 * smooth_noise(rg, u.shape, 1.0 * S + 0.3)
        rc = 0.045
        qx, qy = np.abs(u) - (0.5 - rc), np.abs(v) - (0.5 - rc)
        dbox = np.hypot(np.maximum(qx, 0), np.maximum(qy, 0)) + np.minimum(np.maximum(qx, qy), 0) - rc + wig
        px, py = u + 0.5, v + 0.5
        sw_ = 0.068
        segs = [
            # ㅅ : two strokes meeting at the top
            (0.50, 0.110, 0.240, 0.335), (0.50, 0.110, 0.760, 0.335),
            # ㅜ : a bar, and a short stem down from its middle
            (0.175, 0.465, 0.825, 0.465), (0.50, 0.465, 0.50, 0.605),
            # ㅁ : a closed square
            (0.275, 0.715, 0.725, 0.715), (0.725, 0.715, 0.725, 0.885),
            (0.725, 0.885, 0.275, 0.885), (0.275, 0.885, 0.275, 0.715),
        ]
        dg = np.full(u.shape, 9.0, np.float32)
        for ax, ay, bx, by in segs:
            ex, ey = bx - ax, by - ay
            t = np.clip(((px - ax) * ex + (py - ay) * ey) / (ex * ex + ey * ey), 0, 1)
            dg = np.minimum(dg, np.hypot(px - ax - t * ex, py - ay - t * ey))
        dg = dg - sw_ / 2 + wig * 0.8
        px_ = 1.0 / s                                    # one pixel in seal units
        a_box = np.clip(-dbox / (1.6 * px_) + 0.5, 0, 1)
        a_g = np.clip(dg / (1.3 * px_) + 0.5, 0, 1)
        cover = a_box * a_g
        Hl = self.Ht[y0:y1, x0:x1]
        press = 0.5 + 0.5 * np.tanh(0.9 * smooth_noise(rg, u.shape, 14 * S) + 0.5
                                    + 1.2 * (u * 0.5 - v * 0.35))       # pressed a little harder on one side
        squeeze = np.exp(-np.abs(dbox) / 0.03) * 0.25              # paste squeezed to the rim
        take = np.clip((0.75 * Hl + 0.75 * press + squeeze - 0.34 + firm) / 0.30, 0, 1)
        take = take * take * (3 - 2 * take)
        c = ndi.gaussian_filter(cover * (0.10 + 0.90 * take), 0.45 * S + 0.2)
        thick = 0.86 + 0.14 * take
        verm = np.array([0.800, 0.185, 0.115], np.float32)
        region = col[y0:y1, x0:x1]
        plum = region.mean(axis=2, keepdims=True) / 0.9
        ink = verm[None, None, :] * (0.55 + 0.45 * plum) * thick[..., None]
        cc = (0.95 * c)[..., None]
        col[y0:y1, x0:x1] = region * (1 - cc) + ink * cc

    # .................................................. to light
    def render(self):
        up = np.asarray(Image.fromarray(self.Dfix + self.Dc + self.Pp, mode="F").resize(
            (self.W, self.H), Image.BICUBIC), dtype=np.float32).clip(0, None)
        # settled pigment granulates into the hollows; the fibres stand pale in a wash
        g = 1.0 + 0.13 * (0.5 - ndi.gaussian_filter(self.Ht, 0.7)) * 2.0 - 0.02 * np.minimum(self.Flong, 2.0) + 0.01
        dens = up * np.clip(g, 0.45, 1.6) + self.Dd
        self.dens = dens
        m = (1.0 - np.exp(-dens * 0.9))[..., None]
        k_thin = np.array([1.13, 1.00, 0.83], np.float32)       # pine soot, pale: blue-grey
        k_thick = np.array([0.97, 1.00, 1.07], np.float32)      # thick: warm black
        k = k_thin * (1 - m) + k_thick * m
        T = np.exp(-dens[..., None] * k)
        floor = np.array([0.060, 0.052, 0.048], np.float32)
        col = self.paper * T + floor * (1 - T)
        for sl in self.seals:
            self._press_seal(col, *sl)
        return np.clip(col, 0, 1)

    def save(self, path):
        col = self.render()
        img = (col ** (1 / 1.0) * 255.0 + self.rng.uniform(-0.5, 0.5, col.shape)).round().clip(0, 255).astype(np.uint8)
        Image.fromarray(img, "RGB").save(path, dpi=(300, 300))
        return path

def ss(v, a, b):
    t = np.clip((v - a) / (b - a), 0.0, 1.0)
    return t * t * (3.0 - 2.0 * t)


# the line of the back, from the root of the tail down and out of the sheet
SPINE = [(170, 1760), (480, 1880), (820, 2080), (1150, 2390), (1420, 2800), (1600, 3280), (1700, 3800),
         (1740, 4200)]
# half the breadth of the body along that line (distance along it, half-breadth)
BREADTH = [(0, 62), (200, 74), (500, 132), (900, 290), (1400, 520), (1900, 700), (2400, 800), (3200, 840)]
FLOAT = (1575.0, 560.0)


class Whale:
    def __init__(self):
        sp, _ = crom(SPINE, 4.0)
        self.sp = sp
        t = np.gradient(sp, axis=0)
        self.tan = t / (np.hypot(t[:, 0], t[:, 1])[:, None] + 1e-9)
        self.L = np.r_[0, np.cumsum(np.hypot(*np.diff(sp, axis=0).T))]
        self.tree = cKDTree(sp)
        b = np.asarray(BREADTH, np.float64)
        self.hw_of = lambda L: np.interp(L, b[:, 0], b[:, 1])

    def at(self, L, u=0.0):
        """the point at distance L along the back, u half-breadths to the upper side"""
        i = int(np.clip(np.searchsorted(self.L, L), 0, len(self.L) - 1))
        x, y = self.sp[i]
        tx, ty = self.tan[i]
        h = float(self.hw_of(L)) * u
        return x + ty * h, y - tx * h

    def ang(self, L):
        i = int(np.clip(np.searchsorted(self.L, L), 0, len(self.L) - 1))
        return math.atan2(self.tan[i, 1], self.tan[i, 0])

    def fields(self, X, Y):
        d, idx = self.tree.query(np.c_[X.ravel(), Y.ravel()], workers=-1)
        vx = X.ravel() - self.sp[idx, 0]
        vy = Y.ravel() - self.sp[idx, 1]
        sd = self.tan[idx, 1] * vx - self.tan[idx, 0] * vy
        L = self.L[idx]
        hw = self.hw_of(L)
        sgn = np.where(sd >= 0, 1.0, -1.0)
        u = sgn * d / hw
        f = lambda a: a.reshape(X.shape).astype(np.float32)
        return f(L), f(u), f(d - hw), f(hw)


def scar(sh, pts, width, keep, seed=0):
    """a scar is chosen before the wash, like any paper to be kept: less ink is taken there"""
    S = sh.S
    xy, _ = crom(np.asarray(pts, np.float64) * S, 1.0)
    n = len(xy)
    R = int(width * S * 2) + 4
    x0 = max(0, int(xy[:, 0].min()) - R)
    x1 = min(sh.W, int(xy[:, 0].max()) + R)
    y0 = max(0, int(xy[:, 1].min()) - R)
    y1 = min(sh.H, int(xy[:, 1].max()) + R)
    x0 -= x0 % 2
    y0 -= y0 % 2
    x1 -= x1 % 2
    y1 -= y1 % 2
    if x1 <= x0 or y1 <= y0:
        return
    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
    d, idx = cKDTree(xy).query(np.c_[xx.ravel(), yy.ravel()])
    d = d.reshape(xx.shape)
    t = (idx / (n - 1.0)).reshape(xx.shape)
    rg = np.random.default_rng(sh.seed + 6151 * (seed + 1))
    wv = ndi.gaussian_filter1d(rg.standard_normal(n), max(2.0, 14 * S))
    wv = 1.0 + 0.45 * wv / (np.abs(wv).max() + 1e-9)
    w = width * S * np.sin(np.pi * np.clip(t, 0, 1)) ** 0.6 * wv[idx].reshape(xx.shape)
    a = np.clip((w - d) / max(0.9 * S, 0.7) + 0.5, 0, 1)
    k = (1.0 - (1.0 - keep) * a).astype(np.float32)
    sh.Rfull[y0:y1, x0:x1] *= k
    ks = k.reshape((y1 - y0) // 2, 2, (x1 - x0) // 2, 2).mean(axis=(1, 3))
    sy, sx = slice(y0 // 2, y1 // 2), slice(x0 // 2, x1 // 2)
    sh.Rsim[sy, sx] *= ks
    kk = np.clip((ks - keep) / (1.0 - keep + 1e-6), 0, 1) ** 2 * 0.70 + 0.30
    sh.KX[sy, sx] *= kk
    sh.KY[sy, sx] *= kk


def flick(sh, x, y, r, water, lift, seed=0, squash=1.0, ang=0.0):
    """clean water flicked into a damp wash: where it lands it floats the settled ink again, and
    the water running out from it carries that ink to the rim of the bloom"""
    S = sh.S
    cx, cy, rr = x * S / 2, y * S / 2, r * S / 2
    R = int(rr * 2.7) + 3
    x0, x1 = max(0, int(cx) - R), min(sh.sw, int(cx) + R)
    y0, y1 = max(0, int(cy) - R), min(sh.sh, int(cy) + R)
    if x1 <= x0 or y1 <= y0:
        return
    yy, xx = np.mgrid[y0:y1, x0:x1].astype(np.float32)
    dx, dy = xx - cx, yy - cy
    ca, sa = math.cos(ang), math.sin(ang)
    uu = dx * ca + dy * sa
    vv = (-dx * sa + dy * ca) / squash
    th = np.arctan2(vv, uu)
    rg = np.random.default_rng(sh.seed + 92821 * (seed + 1))
    k = np.zeros_like(th)
    for m in range(2, 10):
        k += rg.standard_normal() / m ** 0.8 * np.cos(m * th + rg.uniform(0, 6.28))
    n2 = 0.65 * smooth_noise(rg, uu.shape, rr / 5.0 + 0.5) + 0.45 * smooth_noise(rg, uu.shape, rr / 16.0 + 0.4)
    d = np.hypot(uu, vv) / (rr * np.clip(1 + 0.30 * k, 0.4, 1.8)) - 0.30 * n2 * np.clip(np.hypot(uu, vv) / rr, 0, 1)
    a = np.clip((1.0 - d) / 0.40, 0, 1)
    a = a * a * (3 - 2 * a)
    a = a * sh.Rsim[y0:y1, x0:x1]
    sl = (slice(y0, y1), slice(x0, x1))
    sh.Ww[sl] = np.minimum(sh.Ww[sl] + water * a, 1.6)
    mv = lift * a * sh.Dc[sl]
    sh.Dc[sl] -= mv
    sh.Pp[sl] += mv


def paint(S, out):
    sh = Sheet(BASE_W, BASE_H, S, SEED)
    rng = np.random.default_rng(SEED + 1)
    wh = Whale()
    X, Y = sh.grid()
    L, u, e, hw = wh.fields(X, Y)
    nz = lambda px: smooth_noise(rng, X.shape, px * S / 2 + 0.3)
    hand = np.clip(1.0 + 0.16 * nz(110) + 0.10 * nz(34) + 0.05 * nz(9), 0.55, 1.5)
    up = ss(u, -0.4, 0.4)                                  # 1 on the upper side of the back, 0 on the lower

    # ---- paper to be kept is chosen first: the old rake scars on the flank
    for j, (L0, u0, ln, sl) in enumerate([(1150, 0.46, 300, 0.12), (1840, -0.52, 400, -0.08)]):
        for i in range(3):
            a = wh.at(L0 + 14 * i, u0 + 0.065 * i * (1500.0 / (L0 + 400)))
            b = wh.at(L0 + ln * (0.9 + 0.12 * i) + 14 * i, u0 + sl + 0.065 * i * (1500.0 / (L0 + 400)))
            m = ((a[0] + b[0]) / 2 + rng.uniform(-9, 9), (a[1] + b[1]) / 2 + rng.uniform(-9, 9))
            scar(sh, [a, m, b], 6.0 - 1.0 * i, 0.45 + 0.08 * i, seed=10 * j + i)

    # ---- pale ink, on wet paper: the water the whale is in. It is wet well beyond the body below
    #      and at the deep end; toward the tail the paper is left dry, so the wash will stop there.
    reach_lo = -45 + 600 * ss(L, 520, 1250)
    reach_up = -45 + 400 * ss(L, 900, 1900)
    reach = reach_lo * (1 - up) + reach_up * up
    wn = 0.6 * nz(150) + 0.4 * nz(45)
    wet = 1.0 - ss(e - reach + 85 * wn, -26, 26)
    wet *= ss(Y, 1500, 1800)
    far = ss(e, 0, 520)
    sh.lay(wet * hand * (0.050 + 0.075 * ss(Y, 2300, 3800)) * (1.0 - 0.45 * far), 0.52 * wet)
    sh.flow(500, evap=0.00025, dp=0.8)

    # ---- middle ink: the whale, one wash.  The brush is full at the bottom of the sheet and
    #      empty by the tail.
    en = 0.55 * nz(70) + 0.45 * nz(18)
    soft = 22 + 120 * ss(L, 700, 1700) * (1.0 - 0.35 * up)
    body = 1.0 - ss(e + 26 * en * ss(L, 300, 900), -0.75 * soft, 0.45 * soft)
    along = np.interp(L, [0, 350, 800, 1300, 1900, 2500, 3300], [0.34, 0.42, 0.62, 0.92, 1.18, 1.36, 1.46])
    across = 0.42 + 0.58 * (1.0 - np.clip(np.abs(u - 0.10), 0, 1) ** 1.5)
    # first the broad dry drags of the big brush, heel at the deep end, running out toward the tail
    for i, (u0, wd, ld, sd_) in enumerate([(0.05, 0.50, 1.00, 0), (-0.50, 0.30, 0.90, 1), (0.60, 0.28, 0.86, 2)]):
        Ls = np.linspace(2300 - 500 * abs(u0), 200 + 260 * abs(u0), 22)
        pts = [wh.at(l, u0 * (0.55 + 0.45 * ss(l, 300, 1500))) for l in Ls]
        rad = [(k / 21.0, max(18.0, wd * float(wh.hw_of(l)))) for k, l in enumerate(Ls)]
        sh.stroke(pts, rad, [(0, 0.80 * ld), (0.35, 0.62 * ld), (0.65, 0.47 * ld), (0.85, 0.37 * ld), (1, 0.27)],
                  conc=0.30, water=0.3, wet=0.5, nb=70, soft=8.0, tooth=0.75, body=0.5, wobble=5.0,
                  streak=1.3, seed=100 + sd_)
    sh.lay(body * along * across * hand, 0.72 * body)

    # the fluke: two strokes out from the root of the tail, each ending dry
    tx, ty = wh.tan[0]
    ox, oy = SPINE[0]
    fl = lambda al, pe: (ox + tx * al + ty * pe, oy + ty * al - tx * pe)       # along the back / to the upper side
    half = [(130, 60), (62, 142), (6, 262), (-70, 392), (-216, 508), (-190, 410), (-182, 290), (-178, 160),
            (-162, 60), (-118, 0)]
    poly = [fl(a, b) for a, b in half] + [fl(a, -b) for a, b in half[-2::-1]]
    sh.patch(poly, 0.40, 0.62, seed=5, rough=0.25)
    for sg, sd_ in ((1, 120), (-1, 121)):
        sh.stroke([fl(-176, 452 * sg), fl(-214, 500 * sg), fl(-252, 534 * sg), fl(-292, 562 * sg)],
                  [(0, 17), (0.35, 15), (0.7, 9), (1, 2)],
                  [(0, 0.50), (0.4, 0.42), (1, 0.25)],
                  conc=0.40, water=0.0, wet=0.0, nb=56, soft=4.0, tooth=0.6, body=0.6, wobble=1.5, streak=0.5,
                  seed=sd_)
    sh.flow(650, evap=0.00025, dp=0.9)

    # ---- dark ink dropped along the spine while the wash is wet, and left to crawl
    band = np.exp(-((u - 0.10 + 0.07 * nz(120)) / 0.23) ** 2) * ss(L, 420, 1300) * (0.7 + 1.2 * ss(L, 1200, 2700))
    band *= np.clip(1.0 + 0.55 * nz(60) + 0.35 * nz(16), 0.1, 2.2) * ss(-e, 0, 30)
    sh.lay(band, 0.30 * np.clip(band, 0, 1))
    for i, (l, r, p) in enumerate([(1500, 105, 2.0), (1790, 150, 2.5), (2110, 190, 2.8), (2440, 230, 3.0),
                                   (2800, 250, 3.0), (1250, 70, 1.5)]):
        x, y = wh.at(l, 0.10 + rng.uniform(-0.05, 0.05))
        sh.drop(x, y, r, p, 0.55, ragged=0.32, seed=40 + i, squash=0.55, ang=wh.ang(l))
    sh.flow(900, evap=0.0007, dp=1.0, carry=0.5)

    # ---- clean water flicked into the damp wash: it pushes the ink out and leaves pale blooms
    for i, (l, uu, r, wt, lf, sq) in enumerate([(1660, 0.56, 170, 0.5, 0.24, 0.45), (2100, -0.50, 220, 0.5, 0.28, 0.40),
                                                (2520, 0.62, 130, 0.55, 0.32, 0.55)]):
        x, y = wh.at(l, uu)
        flick(sh, x, y, r, wt, lf, seed=60 + i, squash=sq, ang=wh.ang(l) + rng.uniform(-0.4, 0.4))
    sh.flow(650, evap=0.0004, dp=0.9, carry=1.0, lift=2.2)
    # and a last few touches of dark into the damp: these hardly move, and keep an edge
    for i, (l, uu, r, p) in enumerate([(1660, 0.02, 60, 2.2), (1950, 0.20, 82, 2.6), (2300, 0.04, 105, 2.8),
                                       (2660, 0.22, 120, 2.8), (2180, -0.10, 50, 2.0)]):
        x, y = wh.at(l, uu)
        sh.drop(x, y, r, p, 0.22, ragged=0.4, seed=70 + i, squash=0.45, ang=wh.ang(l) + rng.uniform(-0.2, 0.2))
    sh.flow(900, evap=0.0004, dp=0.9, carry=1.0, lift=1.2)
    sh.dry()

    # ---- dark ink, dry.  The knuckles of the back, down to the tail.
    Ls = np.linspace(1260, 190, 18)
    kn = [(0, 3.0)]
    for k in range(1, 11):
        t = (k / 11.0) ** 0.85
        kn += [(t - 0.030, 12.0 + 24.0 * (1 - t) ** 0.7), (t + 0.024, 7.0 + 9.0 * (1 - t))]
    kn += [(1, 1.0)]
    sh.stroke([wh.at(l, 0.10 * ss(l, 300, 900)) for l in Ls], kn,
              [(0, 0.30), (0.15, 0.43), (0.5, 0.40), (0.8, 0.36), (1, 0.27)], conc=0.95, nb=30, soft=1.5,
              tooth=0.8, body=0.9, wobble=1.5, streak=1.0, seed=130)

    # ---- at the top: the surface, the rope, me.
    fx, fy = FLOAT
    chop = dict(conc=1.5, nb=12, soft=1.0, tooth=0.7, body=0.9, wobble=1.0, streak=0.5)
    for i, (x0, x1, yy, w) in enumerate([(fx - 470, fx - 215, fy + 34, 5.0), (fx + 150, fx + 330, fy - 30, 4.0),
                                         (fx - 250, fx - 120, fy - 52, 3.0)]):
        xs = np.linspace(x0, x1, 6)
        sh.stroke([(x, yy - 16 * math.sin(math.pi * (x - x0) / (x1 - x0)) + 4 * math.sin(i * 2 + x / 40.0))
                   for x in xs],
                  [(0, 0.4), (0.3, w), (0.7, w * 0.8), (1, 0.4)], [(0, 0.46), (0.4, 0.38), (1, 0.24)],
                  seed=80 + i, **chop)
    # the rope: full at the float, running dry on the way down
    sh.stroke([(fx + 3, fy + 82), (fx - 10, fy + 250), (fx - 2, fy + 450), (fx + 26, fy + 640), (fx + 34, fy + 760)],
              [(0, 2.4), (0.3, 4.4), (0.7, 3.2), (1, 1.2)],
              [(0, 0.85), (0.3, 0.8), (0.48, 0.40), (0.56, 0.7), (0.75, 0.38), (1, 0.22)],
              tone=[(0, 1.0), (0.6, 0.9), (1, 0.6)],
              conc=3.0, water=0.3, wet=0.3, nb=10, soft=1.0, tooth=0.55, body=0.8, wobble=0.6, streak=0.5, seed=60)
    # me: going down head first, five strokes
    ox_, oy_ = fx - 205, fy + 215                                  # her hips
    dn, pp = (-0.47, 0.883), (0.883, 0.47)
    P = lambda a, b: (ox_ + dn[0] * a + pp[0] * b, oy_ + dn[1] * a + pp[1] * b)
    dk = dict(conc=3.2, water=0.3, wet=0.35, nb=10, soft=1.0, tooth=0.5, body=0.85, wobble=0.0, streak=0.6)
    sh.stroke([P(-6, 0), P(26, 1), P(54, 0), P(68, -1), P(88, -1)],
              [(0, 5), (0.25, 12.5), (0.55, 10.5), (0.70, 6.0), (0.83, 11.0), (0.93, 9.5), (1, 3)],
              [(0, 0.9), (1, 0.85)], seed=90, **dk)                                   # back, neck, head
    sh.stroke([P(2, 3), P(-46, 13), P(-98, 12), P(-112, 20)], [(0, 7.5), (0.5, 5.5), (0.9, 2.6), (1, 1.2)],
              [(0, 0.85), (0.6, 0.62), (1, 0.34)], seed=91, **dk)                       # leg
    sh.stroke([P(2, -3), P(-50, -9), P(-94, -24), P(-110, -22)], [(0, 7.5), (0.5, 5.5), (0.9, 2.6), (1, 1.2)],
              [(0, 0.85), (0.6, 0.62), (1, 0.34)], seed=92, **dk)                       # leg
    sh.stroke([P(50, 8), P(88, 22), P(128, 24)], [(0, 5.5), (0.5, 4.4), (1, 1.4)],
              [(0, 0.8), (1, 0.38)], seed=93, **dk)                                     # arm, reaching down
    sh.stroke([P(50, -9), P(30, -17), P(8, -19)], [(0, 5), (0.5, 3.8), (1, 1.4)],
              [(0, 0.78), (1, 0.36)], seed=94, **dk)                                    # the other
    sh.flow(300, evap=0.0012, carry=0.6, dp=0.2)
    sh.dry()

    # ---- the seal, last, where the float is
    sh.seal(fx, fy, 138, rot=-2.5, seed=1, firm=0.26)
    sh.save(out)
    d = sh.dens
    print("bare or nearly bare: %.0f%%   top third inked: %.2f%%" % (
        100.0 * float((d < 0.10).mean()), 100.0 * float((d[:sh.H // 3] > 0.10).mean())))
    return sh
