# =====================================================================
#  THE BRUSH
#  A stroke is four decisions: direction, load, pressure, and where it
#  runs dry.  Everything below is those four, written down.

@njit(cache=True, fastmath=True, error_model='numpy')
def _stroke(col, hgt, tooth, sx, sy, n, r, c0, sv, bl, bc, bend,
            opac, thick, dry, drysens, pickup, edge, ridge):
    H = hgt.shape[0]; W = hgt.shape[1]
    nb = bl.shape[0]
    seg = np.empty(n); cum = np.empty(n); cum[0] = 0.0
    minx = 1e9; maxx = -1e9; miny = 1e9; maxy = -1e9
    for i in range(n):
        if sx[i] < minx: minx = sx[i]
        if sx[i] > maxx: maxx = sx[i]
        if sy[i] < miny: miny = sy[i]
        if sy[i] > maxy: maxy = sy[i]
        if i < n - 1:
            dx = sx[i + 1] - sx[i]; dy = sy[i + 1] - sy[i]
            seg[i] = math.sqrt(dx * dx + dy * dy) + 1e-6
            cum[i + 1] = cum[i] + seg[i]
    L = cum[n - 1] + 1e-6
    # the brush picks up what is already wet on the canvas as it travels;
    # pigments meet as pigments (log mix), not as light
    cs = np.empty((n, 3))
    cur0 = c0[0]; cur1 = c0[1]; cur2 = c0[2]
    for i in range(n):
        if pickup > 0.0 and i > 0:
            xi = int(sx[i]); yi = int(sy[i])
            if xi >= 0 and xi < W and yi >= 0 and yi < H:
                cur0 = math.exp((1 - pickup) * math.log(max(cur0, 0.004)) + pickup * math.log(max(col[yi, xi, 0], 0.004)))
                cur1 = math.exp((1 - pickup) * math.log(max(cur1, 0.004)) + pickup * math.log(max(col[yi, xi, 1], 0.004)))
                cur2 = math.exp((1 - pickup) * math.log(max(cur2, 0.004)) + pickup * math.log(max(col[yi, xi, 2], 0.004)))
        cs[i, 0] = cur0; cs[i, 1] = cur1; cs[i, 2] = cur2
    x0 = max(0, int(minx - r - 2)); x1 = min(W, int(maxx + r + 3))
    y0 = max(0, int(miny - r - 2)); y1 = min(H, int(maxy + r + 3))
    rr = r * r * 1.1
    for y in range(y0, y1):
        for x in range(x0, x1):
            best = 1e18; bi = 0; bt = 0.0; cr = 0.0
            for i in range(n - 1):
                ax = x - sx[i]; ay = y - sy[i]
                bx = sx[i + 1] - sx[i]; by = sy[i + 1] - sy[i]
                t = (ax * bx + ay * by) / (seg[i] * seg[i])
                if t < 0.0: t = 0.0
                elif t > 1.0: t = 1.0
                ex = ax - t * bx; ey = ay - t * by
                d2 = ex * ex + ey * ey
                if d2 < best:
                    best = d2; bi = i; bt = t; cr = bx * ay - by * ax
            if best > rr:
                continue
            tau = (cum[bi] + bt * seg[bi]) / L
            wf = 0.93 + 0.07 * math.sin(3.14159 * min(1.0, 0.12 + tau))   # a flat brush: the width barely changes
            dd = math.sqrt(best); over = 0.0
            if bi == 0 and bt <= 0.0:                                     # beyond the heel of the stroke
                over = -((x - sx[0]) * (sx[1] - sx[0]) + (y - sy[0]) * (sy[1] - sy[0])) / seg[0]
                dd = abs(cr) / seg[0]
            elif bi == n - 2 and bt >= 1.0:                               # beyond the toe
                over = ((x - sx[n - 1]) * (sx[n - 1] - sx[n - 2]) + (y - sy[n - 1]) * (sy[n - 1] - sy[n - 2])) / seg[n - 2]
                dd = abs(cr) / seg[n - 2]
            q = dd / (r * wf)
            if q >= 1.0:
                continue
            e = 1.0
            if q > 1.0 - edge:
                e = (1.0 - q) / edge
                e = e * e * (3 - 2 * e)
            if cr < 0: sq = -q
            else: sq = q
            k = int((sq * 0.5 + 0.5) * (nb - 1) + 0.5)
            if k < 0: k = 0
            if k >= nb: k = nb - 1
            if over > r * 0.5 * (bl[k] - 0.5):                            # square end, ragged by the bristles
                continue
            # each bristle lifts off at its own moment
            ce = (1.0 - bend[k] - tau) / 0.05
            if ce <= 0.0: continue
            if ce > 1.0: ce = 1.0
            cb = (tau + 0.02 - 0.3 * bend[nb - 1 - k]) / 0.04
            if cb <= 0.0: continue
            if cb > 1.0: cb = 1.0
            # the load runs out; a dry brush only touches the tops of the weave
            avail = bl[k] * (1.0 - dry * tau ** 1.5)
            cov = (avail - drysens * (1.0 - tooth[y, x]) - 0.08) / 0.2
            if cov <= 0.0: continue
            if cov > 1.0: cov = 1.0
            a = opac * e * cov * ce * cb
            f = bc[k]
            c_r = cs[bi, 0] + bt * (cs[bi + 1, 0] - cs[bi, 0]) + f * sv[0]
            c_g = cs[bi, 1] + bt * (cs[bi + 1, 1] - cs[bi, 1]) + f * sv[1]
            c_b = cs[bi, 2] + bt * (cs[bi + 1, 2] - cs[bi, 2]) + f * sv[2]
            col[y, x, 0] += a * (c_r - col[y, x, 0])
            col[y, x, 1] += a * (c_g - col[y, x, 1])
            col[y, x, 2] += a * (c_b - col[y, x, 2])
            bump = (q - 0.84) / 0.11
            hd = thick * (0.4 + 0.6 * bl[k]) * (1.0 - 0.45 * tau) + ridge * thick * math.exp(-bump * bump)
            hgt[y, x] = hgt[y, x] * (1.0 - 0.6 * a) + a * hd


@njit(cache=True, fastmath=True, error_model='numpy')
def _bristles(r, bl, bc, bend):
    nb = int(2 * r / 1.7)
    if nb < 4: nb = 4
    if nb > bl.shape[0]: nb = bl.shape[0]
    w = np.random.randn(); l = np.random.rand()
    for k in range(nb):
        w = 0.6 * w + 0.8 * np.random.randn()
        l = 0.5 * l + 0.5 * np.random.rand()
        bc[k] = max(-1.5, min(1.5, w))
        bl[k] = 0.55 + 0.45 * l
        bend[k] = 0.28 * np.random.rand() ** 2
    return nb


@njit(cache=True, fastmath=True, error_model='numpy')
def _trace(src, ang, reg, x, y, dx, dy, step, nmax, c0, tol, rg, ox, oy, wander):
    H = ang.shape[0]; W = ang.shape[1]
    n = 0
    for i in range(nmax):
        x += dx * step; y += dy * step
        xi = int(x); yi = int(y)
        if xi < 1 or xi >= W - 1 or yi < 1 or yi >= H - 1: break
        if reg[yi, xi] != rg: break
        d = abs(src[yi, xi, 0] - c0[0]) + abs(src[yi, xi, 1] - c0[1]) + abs(src[yi, xi, 2] - c0[2])
        if d > tol: break
        ox[n] = x; oy[n] = y; n += 1
        a = ang[yi, xi] + wander * np.random.randn()
        nx = math.cos(a); ny = math.sin(a)
        if nx * dx + ny * dy < 0: nx = -nx; ny = -ny
        dx = 0.55 * dx + 0.45 * nx; dy = 0.55 * dy + 0.45 * ny
        m = math.sqrt(dx * dx + dy * dy) + 1e-9
        dx /= m; dy /= m
    return n


@njit(cache=True, fastmath=True, error_model='numpy')
def _layer(col, hgt, tooth, src, ang, reg, lenf, px, py, pr, seed, maxlen, tol, cjit, streak,
           opac, thick, thickv, dry, drysens, pickup, edge, ridge, wander):
    np.random.seed(seed)
    fx = np.empty(160); fy = np.empty(160); gx = np.empty(160); gy = np.empty(160)
    sx = np.empty(330); sy = np.empty(330)
    bl = np.empty(128); bc = np.empty(128); bend = np.empty(128)
    c = np.empty(3); sv = np.empty(3); c0 = np.empty(3)
    for j in range(px.shape[0]):
        x = px[j]; y = py[j]; r = pr[j]
        xi = int(x); yi = int(y)
        c0[0] = src[yi, xi, 0]; c0[1] = src[yi, xi, 1]; c0[2] = src[yi, xi, 2]
        rg = reg[yi, xi]
        step = max(1.5, 0.6 * r)
        nmax = int(maxlen * r * lenf[yi, xi] / step)
        if nmax < 1: nmax = 1
        if nmax > 150: nmax = 150
        a0 = ang[yi, xi] + wander * np.random.randn()
        dx = math.cos(a0); dy = math.sin(a0)
        nfw = int(nmax * (0.3 + 0.4 * np.random.rand())) + 1
        nf = _trace(src, ang, reg, x, y, dx, dy, step, nfw, c0, tol, rg, fx, fy, wander)
        ng = _trace(src, ang, reg, x, y, -dx, -dy, step, nmax - nfw + 1, c0, tol, rg, gx, gy, wander)
        n = 0
        for i in range(ng - 1, -1, -1):
            sx[n] = gx[i]; sy[n] = gy[i]; n += 1
        sx[n] = x; sy[n] = y; n += 1
        for i in range(nf):
            sx[n] = fx[i]; sy[n] = fy[i]; n += 1
        if n < 2:
            sx[1] = x + dx * step; sy[1] = y + dy * step; n = 2
        if np.random.rand() < 0.5:
            for i in range(n // 2):
                tx = sx[i]; sx[i] = sx[n - 1 - i]; sx[n - 1 - i] = tx
                ty = sy[i]; sy[i] = sy[n - 1 - i]; sy[n - 1 - i] = ty
        nb = _bristles(r, bl, bc, bend)
        v = 1.0 + 0.3 * cjit * np.random.randn()
        g = np.random.randn()
        h0 = np.random.randn(); h1 = np.random.randn(); h2 = np.random.randn()
        hm = 0.3 * h0 + 0.6 * h1 + 0.1 * h2
        l0 = 0.3 * c0[0] + 0.6 * c0[1] + 0.1 * c0[2]
        c[0] = c0[0] * v + cjit * (h0 - hm) * (0.06 + l0)
        c[1] = c0[1] * v + cjit * (h1 - hm) * (0.06 + l0)
        c[2] = c0[2] * v + cjit * (h2 - hm) * (0.06 + l0)
        for k in range(3):
            c[k] = min(1.0, max(0.0, c[k]))
            sv[k] = streak * (0.5 * g * (0.25 + c[k]) + 0.4 * np.random.randn())
        lum = 0.3 * c[0] + 0.6 * c[1] + 0.1 * c[2]
        th = thick * (1.0 + thickv * (lum - 0.35) * 2.2)
        if th < 0.05 * thick: th = 0.05 * thick
        _stroke(col, hgt, tooth, sx, sy, n, r, c, sv, bl[:nb], bc[:nb], bend[:nb],
                opac, th, dry, drysens, pickup, edge, ridge)


@njit(cache=True, fastmath=True, error_model='numpy')
def _one(col, hgt, tooth, sx, sy, r, c, seed, streak, opac, thick, dry, drysens, pickup, edge, ridge):
    np.random.seed(seed)
    bl = np.empty(128); bc = np.empty(128); bend = np.empty(128); sv = np.empty(3)
    nb = _bristles(r, bl, bc, bend)
    g = np.random.randn()
    for k in range(3):
        sv[k] = streak * (0.6 * g * (0.25 + c[k]) + 0.35 * np.random.randn())
    _stroke(col, hgt, tooth, sx, sy, sx.shape[0], r, c, sv, bl[:nb], bc[:nb], bend[:nb],
            opac, thick, dry, drysens, pickup, edge, ridge)


@njit(cache=True, fastmath=True, error_model='numpy')
def _ribbon(px, py, pw, W, H):
    """distance field of a tapering polyline: returns crop of (q, t, angle)."""
    n = px.shape[0]
    wm = pw.max()
    x0 = max(0, int(px.min() - wm - 2)); x1 = min(W, int(px.max() + wm + 3))
    y0 = max(0, int(py.min() - wm - 2)); y1 = min(H, int(py.max() + wm + 3))
    if x1 <= x0 or y1 <= y0:
        return 0, 0, np.full((1, 1), 9.0, np.float32), np.zeros((1, 1), np.float32), np.zeros((1, 1), np.float32)
    q = np.full((y1 - y0, x1 - x0), 9.0, np.float32)
    tt = np.zeros((y1 - y0, x1 - x0), np.float32)
    aa = np.zeros((y1 - y0, x1 - x0), np.float32)
    for y in range(y0, y1):
        for x in range(x0, x1):
            best = 1e18; bi = 0; bt = 0.0; cr = 0.0
            for i in range(n - 1):
                ax = x - px[i]; ay = y - py[i]
                bx = px[i + 1] - px[i]; by = py[i + 1] - py[i]
                t = (ax * bx + ay * by) / (bx * bx + by * by + 1e-9)
                if t < 0.0: t = 0.0
                elif t > 1.0: t = 1.0
                ex = ax - t * bx; ey = ay - t * by
                d2 = ex * ex + ey * ey
                if d2 < best:
                    best = d2; bi = i; bt = t; cr = bx * ay - by * ax
            w = pw[bi] + bt * (pw[bi + 1] - pw[bi])
            d = math.sqrt(best) / (w + 1e-6)
            if d < 3.0:
                q[y - y0, x - x0] = d if cr >= 0 else -d
                tt[y - y0, x - x0] = (bi + bt) / (n - 1)
                aa[y - y0, x - x0] = math.atan2(py[bi + 1] - py[bi], px[bi + 1] - px[bi])
    return x0, y0, q, tt, aa


@njit(cache=True, fastmath=True, parallel=True, error_model='numpy')
def _lic(col, ang, mask, L):
    """drag the wet paint along the way the brush went."""
    H = col.shape[0]; W = col.shape[1]
    out = col.copy()
    for y in prange(H):
        for x in range(W):
            m = mask[y, x]
            if m < 0.03: continue
            n = int(L * m) + 1
            a0 = col[y, x, 0]; a1 = col[y, x, 1]; a2 = col[y, x, 2]; cnt = 1.0
            for sg in range(2):
                a = ang[y, x]
                dx = math.cos(a); dy = math.sin(a)
                if sg == 1: dx = -dx; dy = -dy
                fx = x + 0.0; fy = y + 0.0
                for k in range(n):
                    fx += dx; fy += dy
                    xi = int(fx + 0.5); yi = int(fy + 0.5)
                    if xi < 0 or xi >= W or yi < 0 or yi >= H: break
                    w = 1.0 - k / (n + 1.0)
                    a0 += w * col[yi, xi, 0]; a1 += w * col[yi, xi, 1]; a2 += w * col[yi, xi, 2]; cnt += w
                    b = ang[yi, xi]; nx = math.cos(b); ny = math.sin(b)
                    if nx * dx + ny * dy < 0: nx = -nx; ny = -ny
                    dx = nx; dy = ny
            out[y, x, 0] = a0 / cnt; out[y, x, 1] = a1 / cnt; out[y, x, 2] = a2 / cnt
    return out

def smooth(a, b, x):
    t = np.clip((x - a) / (b - a), 0.0, 1.0)
    return t * t * (3 - 2 * t)

def lerp(a, b, t):
    a = np.asarray(a, np.float32); b = np.asarray(b, np.float32)
    if np.ndim(t) == 2 and np.shape(t)[-1] != 1 and (a.ndim == 1 or b.ndim == 1 or a.ndim == 3):
        t = t[..., None]
    return a + (b - a) * t

def vnoise(H, W, cell, rng, aspect=1.0):
    """smooth noise in 0..1; features `cell` px wide and cell*aspect px tall."""
    cx = max(1.0, cell); cy = max(1.0, cell * aspect)
    gh = int(H / cy) + 4; gw = int(W / cx) + 4
    g = rng.random((gh, gw)).astype(np.float32)
    im = Image.fromarray(g).resize((int(gw * cx) + 1, int(gh * cy) + 1), Image.BICUBIC)
    a = np.asarray(im)
    oy = int(cy * (1 + rng.random())); ox = int(cx * (1 + rng.random()))
    return np.clip(a[oy:oy + H, ox:ox + W], 0, 1)

def fbm(H, W, cell, rng, octaves=4, aspect=1.0, gain=0.5):
    out = np.zeros((H, W), np.float32); amp = 1.0; tot = 0.0
    for o in range(octaves):
        out += amp * vnoise(H, W, cell / 2 ** o, rng, aspect); tot += amp; amp *= gain
    out /= tot
    return np.clip((out - 0.5) * 1.9 + 0.5, 0, 1)

def blur(a, s):
    if a.ndim == 3:
        return np.stack([gaussian_filter(a[..., k], s) for k in range(3)], -1)
    return gaussian_filter(a, s)

def lum(c):
    return 0.3 * c[..., 0] + 0.6 * c[..., 1] + 0.1 * c[..., 2]

def spline(pts, n=40):
    pts = np.asarray(pts, float)
    d = np.r_[0, np.cumsum(np.hypot(*np.diff(pts, axis=0).T))]
    cs = CubicSpline(d, pts, bc_type='natural') if len(pts) > 2 else None
    t = np.linspace(0, d[-1], n)
    return cs(t) if cs is not None else np.stack([np.interp(t, d, pts[:, 0]), np.interp(t, d, pts[:, 1])], 1)

class Painting:
    def __init__(self, W, H, seed, ground=(0.60, 0.33, 0.20)):
        self.W, self.H = W, H
        self.S = W / 2400.0
        self.rng = np.random.default_rng(seed)
        self.seed = seed
        S, rng = self.S, self.rng
        Y, X = np.mgrid[0:H, 0:W].astype(np.float32)
        self.X, self.Y = X, Y
        # coarse canvas: plain weave, uneven threads
        p = 7.0 * S if S >= 0.6 else 5.0
        wx = (vnoise(H, W, 90 * S, rng) - 0.5) * 5 * S
        wy = (vnoise(H, W, 90 * S, rng) - 0.5) * 5 * S
        wv = 0.5 + 0.5 * np.cos(2 * np.pi * (X + wx) / p) * np.cos(2 * np.pi * (Y + wy) / p)
        slub = vnoise(H, W, 5 * S, rng, aspect=30) * 0.5 + vnoise(H, W, 150 * S, rng, aspect=0.03) * 0.5
        self.tooth = np.clip(wv * (0.75 + 0.5 * slub), 0, 1).astype(np.float32)
        # toned ground, scrubbed on with a wide brush
        g = np.asarray(ground, np.float32)
        scr = fbm(H, W, 260 * S, rng, 3, aspect=0.25)
        self.col = (g[None, None, :] * (0.86 + 0.24 * scr[..., None]) * (0.93 + 0.07 * self.tooth[..., None])).astype(np.float32)
        self.ground = self.col.copy()
        self.hgt = np.zeros((H, W), np.float32)
        self.ref = np.zeros((H, W, 3), np.float32)
        self.ang = np.zeros((H, W), np.float32)
        self.lenf = np.ones((H, W), np.float32)
        self.reg = np.zeros((H, W), np.int32)
        self.detail = np.zeros((H, W), np.float32)
        # the painted field stops short of the edge, unevenly
        d = np.minimum(np.minimum(X, W - 1 - X), np.minimum(Y, H - 1 - Y))
        self.edge = d + (fbm(H, W, 120 * S, rng, 3) - 0.5) * 60 * S
        self._k = 0

    # ---- building the drawing ------------------------------------------------
    def put(self, mask, colour, ang=None, reg=None, lenf=None, detail=None, at=None):
        """lay a shape into the drawing.  at=(x0,y0) for a cropped mask."""
        if at is None:
            sl = (slice(None), slice(None))
        else:
            sl = (slice(at[1], at[1] + mask.shape[0]), slice(at[0], at[0] + mask.shape[1]))
        m = np.clip(mask, 0, 1).astype(np.float32)
        c = np.asarray(colour, np.float32)
        self.ref[sl] += m[..., None] * (c - self.ref[sl])
        hard = m > 0.5
        if ang is not None:
            self.ang[sl] = np.where(hard, ang, self.ang[sl])
        if reg is not None:
            self.reg[sl] = np.where(hard, reg, self.reg[sl])
        if lenf is not None:
            self.lenf[sl] = np.where(hard, lenf, self.lenf[sl])
        if detail is not None:
            self.detail[sl] = np.maximum(self.detail[sl], m * detail)

    def ribbon(self, pts, widths):
        pts = np.asarray(pts, np.float64); w = np.asarray(widths, np.float64)
        if w.ndim == 0: w = np.full(len(pts), float(w))
        if len(w) != len(pts): w = np.interp(np.linspace(0, 1, len(pts)), np.linspace(0, 1, len(w)), w)
        return _ribbon(np.ascontiguousarray(pts[:, 0]), np.ascontiguousarray(pts[:, 1]), w, self.W, self.H)

    # ---- painting ------------------------------------------------------------
    def layer(self, src, r, spacing=0.9, mask=None, err=None, order='value', maxlen=6.0, tol=0.22,
              cjit=0.05, streak=0.05, opac=0.95, thick=1.0, thickv=0.6, dry=0.6, drysens=0.5,
              pickup=0.06, edge=0.35, ridge=0.4, wander=0.12, sizev=0.3, margin=26):
        W, H, S, rng = self.W, self.H, self.S, self.rng
        r = r * S
        g = max(1.5, r * spacing)
        xs = np.arange(g / 2, W, g); ys = np.arange(g / 2, H, g)
        gx, gy = np.meshgrid(xs, ys)
        px = (gx + rng.uniform(-g / 2, g / 2, gx.shape)).ravel().clip(1, W - 2)
        py = (gy + rng.uniform(-g / 2, g / 2, gy.shape)).ravel().clip(1, H - 2)
        ix = px.astype(int); iy = py.astype(int)
        keep = self.edge[iy, ix] > margin * S
        if mask is not None:
            keep &= rng.random(len(px)) < mask[iy, ix]
        if err is not None:
            e = np.abs(blur(self.col, max(1.0, r * 0.5)) - src).sum(-1)
            keep &= e[iy, ix] > err / (0.35 + 1.5 * self.detail[iy, ix])
        px, py, ix, iy = px[keep], py[keep], ix[keep], iy[keep]
        if len(px) == 0: return 0
        if order == 'value':      # darks first and thin, lights last and thick
            o = np.argsort(lum(src[iy, ix]) + rng.normal(0, 0.10, len(px)))
        else:
            o = rng.permutation(len(px))
        px, py = px[o], py[o]
        pr = r * (1.0 + sizev * rng.uniform(-1, 1, len(px)))
        self._k += 1
        _layer(self.col, self.hgt, self.tooth, np.ascontiguousarray(src, np.float32), self.ang, self.reg, self.lenf,
               px, py, pr, self.seed % 100000 + 17 * self._k, maxlen, tol, cjit, streak,
               opac, thick, thickv, dry, drysens, pickup, edge, ridge, wander)
        return len(px)

    def stroke(self, pts, r, colour, streak=0.04, opac=1.0, thick=1.0, dry=0.5, drysens=0.4,
               pickup=0.0, edge=0.3, ridge=0.4, smooth_pts=True):
        """one deliberate stroke along a path (canvas pixels)."""
        pts = np.asarray(pts, float)
        r = max(0.7, r)
        ln = np.hypot(*np.diff(pts, axis=0).T).sum()
        n = int(max(2, min(300, ln / max(1.2, 0.6 * r) + 1)))
        p = spline(pts, n) if smooth_pts and len(pts) > 2 else np.stack(
            [np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pts)), pts[:, k]) for k in (0, 1)], 1)
        self._k += 1
        _one(self.col, self.hgt, self.tooth, np.ascontiguousarray(p[:, 0]), np.ascontiguousarray(p[:, 1]), float(r),
             np.asarray(colour, np.float64), self.seed % 100000 + 31 * self._k, streak, opac, thick, dry, drysens,
             pickup, edge, ridge)

    def dab(self, x, y, a, length, r, colour, **kw):
        dx, dy = math.cos(a) * length / 2, math.sin(a) * length / 2
        self.stroke([(x - dx, y - dy), (x + dx, y + dy)], r, colour, smooth_pts=False, **kw)

    def glaze(self, mask, colour, strength=0.5, cell=40):
        """a transparent film: darkens and tints, never covers."""
        S, rng = self.S, self.rng
        st = 0.7 + 0.3 * fbm(self.H, self.W, cell * S, rng, 2, aspect=4)
        m = (np.clip(mask, 0, 1) * strength * st)[..., None]
        c = np.asarray(colour, np.float32)
        self.col *= (1 - m) + m * c

    def blend(self, mask, length=14, soften=0.0):
        """work wet into wet where the edges should be lost."""
        m = np.clip(mask, 0, 1).astype(np.float32)
        self.col = _lic(self.col, self.ang, m, float(length * self.S))
        if soften > 0:
            b = blur(self.col, soften * self.S)
            self.col += (m * 0.6)[..., None] * (b - self.col)
        self.hgt *= (1 - 0.5 * m)

    def sign(self, x, y, h, colour=(0.62, 0.20, 0.10)):
        """lamp mark, then the name.  x,y = lower-right corner of the signature, canvas px."""
        S = self.S
        w = 3.94 * h
        x0 = x - w; y0 = y - h
        r = max(1.0, 0.045 * h)
        for path in SIGNATURE:
            p = [(x0 + a * h, y0 + b * h) for a, b in path]
            self.stroke(p, r, colour, streak=0.02, thick=0.5, dry=0.25, drysens=0.25, edge=0.5, smooth_pts=len(p) > 3)
        lx = x0 - 0.75 * h; ly = y0 + 0.70 * h          # the lamp: a bowl, a wick, a flame
        self.stroke([(lx - 0.30 * h, ly - 0.10 * h), (lx - 0.16 * h, ly + 0.08 * h), (lx + 0.16 * h, ly + 0.08 * h),
                     (lx + 0.30 * h, ly - 0.10 * h)], r, colour, streak=0.02, thick=0.5, dry=0.2, drysens=0.25)
        self.stroke([(lx - 0.12 * h, ly + 0.14 * h), (lx + 0.12 * h, ly + 0.14 * h)], r, colour, thick=0.5, dry=0.2, drysens=0.25)
        self.stroke([(lx, ly - 0.16 * h), (lx - 0.07 * h, ly - 0.32 * h), (lx, ly - 0.56 * h), (lx + 0.07 * h, ly - 0.32 * h),
                     (lx, ly - 0.16 * h)], r * 0.9, colour, thick=0.5, dry=0.2, drysens=0.25)

    # ---- the finished surface under a raking light ---------------------------
    def finish(self, relief=1.0, sat=1.03, weave=0.22):
        S = self.S
        h = gaussian_filter(self.hgt, 1.0 * S)
        wv = self.tooth * np.exp(-2.2 * np.clip(h, 0, 3))             # thin paint lets the canvas show
        surf = (1.5 * h + weave * wv) * S * relief
        gy, gx = np.gradient(surf)
        l = np.array([-0.55, -0.62, 0.56]); l /= np.linalg.norm(l)    # window high on the left
        nz = 1.0 / np.sqrt(gx * gx + gy * gy + 1.0)
        shade = (-gx * l[0] - gy * l[1] + l[2]) * nz / l[2]
        hv = np.array([l[0], l[1], l[2] + 1.0]); hv /= np.linalg.norm(hv)
        spec = np.clip((-gx * hv[0] - gy * hv[1] + hv[2]) * nz, 0, 1) ** 60
        spec = (spec - spec.mean()) * np.clip(h, 0, 1.2) * 0.07
        out = self.col * (0.70 + 0.30 * shade[..., None]) + spec[..., None] * np.array([1.0, 0.97, 0.9])
        m = out.mean(-1, keepdims=True)
        out = m + (out - m) * sat
        out += (self.rng.random(out.shape[:2])[..., None] - 0.5) / 255.0
        return np.clip(out, 0, 1)

@njit(cache=True, fastmath=True, error_model='numpy')
def hash3(i, j, k):
    n = (i * 374761393 + j * 668265263 + k * 1274126177) & 0x7fffffff
    n = ((n ^ (n >> 13)) * 1274126177) & 0x7fffffff
    return ((n ^ (n >> 16)) & 0xffffff) / 16777216.0

@njit(cache=True, fastmath=True, error_model='numpy')
def noise3(x, y, z):
    i = math.floor(x); j = math.floor(y); k = math.floor(z)
    u = x - i; v = y - j; w = z - k
    u = u * u * (3 - 2 * u); v = v * v * (3 - 2 * v); w = w * w * (3 - 2 * w)
    i = int(i); j = int(j); k = int(k)
    a = hash3(i, j, k) + u * (hash3(i + 1, j, k) - hash3(i, j, k))
    b = hash3(i, j + 1, k) + u * (hash3(i + 1, j + 1, k) - hash3(i, j + 1, k))
    c = hash3(i, j, k + 1) + u * (hash3(i + 1, j, k + 1) - hash3(i, j, k + 1))
    d = hash3(i, j + 1, k + 1) + u * (hash3(i + 1, j + 1, k + 1) - hash3(i, j + 1, k + 1))
    e = a + v * (b - a); f = c + v * (d - c)
    return e + w * (f - e)

@njit(cache=True, fastmath=True, error_model='numpy')
def fbm3(x, y, z, o):
    s = 0.0; a = 0.5; t = 0.0
    for i in range(o):
        s += a * noise3(x, y, z); t += a; a *= 0.5
        x = x * 2.03 + 11.3; y = y * 2.03 + 5.7; z = z * 2.03 + 3.1
    return s / t

@njit(cache=True, fastmath=True, error_model='numpy')
def fbm2(x, z, o):
    return fbm3(x, 0.37, z, o)

@njit(cache=True, fastmath=True, error_model='numpy')
def sstep(a, b, x):
    t = (x - a) / (b - a)
    if t < 0: t = 0.0
    if t > 1: t = 1.0
    return t * t * (3 - 2 * t)

@njit(cache=True, fastmath=True, error_model='numpy')
def sd_box(x, y, z, bx, by, bz):
    qx = abs(x) - bx; qy = abs(y) - by; qz = abs(z) - bz
    ox = max(qx, 0.0); oy = max(qy, 0.0); oz = max(qz, 0.0)
    return math.sqrt(ox * ox + oy * oy + oz * oz) + min(max(qx, max(qy, qz)), 0.0)

@njit(cache=True, fastmath=True, error_model='numpy')
def sd_capsule(x, y, z, ax, ay, az, bx, by, bz, r):
    px = x - ax; py = y - ay; pz = z - az
    ex = bx - ax; ey = by - ay; ez = bz - az
    h = (px * ex + py * ey + pz * ez) / (ex * ex + ey * ey + ez * ez + 1e-12)
    if h < 0: h = 0.0
    if h > 1: h = 1.0
    px -= ex * h; py -= ey * h; pz -= ez * h
    return math.sqrt(px * px + py * py + pz * pz) - r

@njit(cache=True, fastmath=True, error_model='numpy')
def smin(a, b, k):
    h = max(k - abs(a - b), 0.0) / k
    return min(a, b) - h * h * k * 0.25

@njit(cache=True, fastmath=True, error_model='numpy')
def ground_h(x, z):
    ax = abs(x)
    h = 0.05 * (fbm2(x * 0.7, z * 0.7, 3) - 0.5)
    h -= 0.085 * math.exp(-((x + 0.35) / 1.2) ** 2 - ((z - 4.95) / 0.60) ** 2)      # where the rain will stand
    h -= 0.06 * math.exp(-((x - 1.1) / 0.45) ** 2 - ((z - 5.3) / 0.2) ** 2)
    bank = 0.95 * sstep(6.2, 8.3, z + 0.5 * (fbm2(x * 0.9 + 3.0, 1.7, 2) - 0.5)) + 0.12 * (fbm2(x * 1.4 + 7.0, z * 1.4, 2) - 0.5) * sstep(6.0, 7.2, z)
    if z > 12.3:
        return 0.95 + 0.25 * (fbm2(x * 0.3, z * 0.3, 2) - 0.5) + h
    side = sstep(1.42, 1.62, ax)
    return h + bank * side + (-0.3 * sstep(5.95, 6.0, z)) * (1.0 - side)


@njit(cache=True, fastmath=True, error_model='numpy')
def scene(x, y, z):

    d = (y - ground_h(x, z)) * 0.6; m = 1
    dw = y + 0.022
    if dw < d: d = dw; m = 10
    ax = abs(x)
    # ---- six laterite steps and the platform
    if z > 5.0 and z < 13.5 and ax < 4.0:
        ds = 1e9
        for i in range(6):
            top = 0.17 * (i + 1)
            z0 = 6.0 + 0.42 * i
            b = sd_box(x, y - (top - 1.0) * 0.5 + 0.5 - 0.5, z - (z0 + 12.6) * 0.5, 1.6, (top + 1.0) * 0.5, (12.6 - z0) * 0.5) - 0.012
            if b < ds: ds = b
        b = sd_box(x, y - (PLAT - 1.0) * 0.5, z - 10.52, 3.0, (PLAT + 1.0) * 0.5, 2.0) - 0.012
        if b < ds: ds = b
        if ds < 0.25:
            ds -= 0.010 * (noise3(x * 9.0, y * 9.0, z * 9.0) - 0.5)                  # laterite is pitted
        if ds < d: d = ds; m = 2
    # ---- two great trunks, close
    for k in range(2):
        if k == 0: tx = -2.22; tz = 4.6; r0 = 0.80; ln = 0.035
        else: tx = 2.68; tz = 5.6; r0 = 0.70; ln = -0.03
        dx = x - tx - ln * y; dz = z - tz
        rr = math.sqrt(dx * dx + dz * dz)
        if rr < r0 * 3.2:
            th = math.atan2(dz, dx)
            yy = max(y, 0.0)
            r = r0 * (1.0 + 0.85 * math.exp(-yy / 0.75)) * (1.0 + 0.16 * math.exp(-yy / 1.8) * math.cos(5.0 * th + 1.3 + k))
            r += 0.045 * (fbm3(th * 5.0, y * 0.9, k * 7.0, 3) - 0.5) + 0.015 * math.sin(th * 34.0 + 3.0 * noise3(th * 3.0, y * 0.6, 1.0))
            dt = (rr - r) * 0.55
        else:
            dt = rr - r0 * 2.4
        if dt < d: d = dt; m = 3
    # ---- the rest of the grove
    if z > 12.0:
        for k in range(TRUNKS.shape[0]):
            dx = x - TRUNKS[k, 0] - TRUNKS[k, 3] * y - 0.12 * math.sin(y * 0.35 + k); dz = z - TRUNKS[k, 1]
            rr = math.sqrt(dx * dx + dz * dz)
            r = TRUNKS[k, 2] * (1.0 + 0.6 * math.exp(-max(y - 0.95, 0.0) / 0.5))
            dt = (rr - r) * 0.8
            if dt < d: d = dt; m = 3
    # ---- serpent stones
    if z > 10.1 and z < 11.3 and x > -0.3 and x < 2.8 and y < 2.4:
        for k in range(4):
            cx = STONES[k, 0]; hh = STONES[k, 1]; hw = STONES[k, 2] * 0.5
            px = abs(x - cx); py = y - PLAT
            if py < hh - hw:
                d2 = px - hw
            else:
                d2 = math.sqrt(px * px + (py - (hh - hw)) ** 2) - hw
            dz = abs(z - STONEZ) - 0.075
            dd = max(d2, dz)
            if d2 > 0 and dz > 0: dd = math.sqrt(d2 * d2 + dz * dz)
            dd = (dd - 0.02) * 0.8
            if dd < d: d = dd; m = 4
    else:
        bd = sd_box(x - 1.25, y - 1.2, z - 10.7, 1.55, 1.2, 0.6)        # outside the group: never step further than its box
        if bd < d: d = max(bd, 0.03)
    # ---- the lamp post
    if abs(x - FLX) < 0.5 and abs(z - FLZ) < 0.5 and y < 2.2:
        dx = x - FLX; dz = z - FLZ
        rr = math.sqrt(dx * dx + dz * dz)
        py = y - PLAT
        dd = max(rr - (0.060 + 0.03 * math.exp(-py / 0.08) + 0.012 * math.exp(-((py - 0.45) / 0.03) ** 2)), py - 0.88)
        cup = max(rr - (0.05 + 0.42 * max(py - 0.84, 0.0)), max(0.84 - py, py - 0.975))
        if cup < dd: dd = cup
        dd = max(dd, -py - 0.2)
        if dd < d: d = dd; m = 5
    else:
        bd = sd_box(x - FLX, y - 1.1, z - FLZ, 0.5, 1.1, 0.5)

    # ---- the one who lit it
    if abs(x - FGX - 0.3) < 1.1 and abs(z - FGZ) < 0.7 and y < 2.9:

        sk = max((rr - r) * 0.8, max(py - 0.95, -py))
        to = sd_capsule(x, y, z, FGX + 0.01, PLAT + 0.90, FGZ, FGX + 0.115, PLAT + 1.25, FGZ - 0.01, 0.125)
        to -= 0.012 * math.sin((y * 14.0 - dx * 9.0 + dz * 5.0))                    # the upper cloth, wound across

        dd = cl; mm = 7
        hd = sd_capsule(x, y, z, FGX + 0.175, PLAT + 1.385, FGZ - 0.01, FGX + 0.20, PLAT + 1.44, FGZ - 0.01, 0.088)
        nk = sd_capsule(x, y, z, FGX + 0.13, PLAT + 1.30, FGZ, FGX + 0.17, PLAT + 1.38, FGZ, 0.045)
        a1 = sd_capsule(x, y, z, FGX + 0.13, PLAT + 1.24, FGZ - 0.13, FGX + 0.38, PLAT + 1.10, FGZ - 0.17, 0.040)
        a2 = sd_capsule(x, y, z, FGX + 0.38, PLAT + 1.10, FGZ - 0.17, FLX - 0.115, FLY - 0.075, FLZ - 0.02, 0.031)
        s2 = min(min(hd, nk), smin(a1, a2, 0.02))
        if s2 < dd: dd = s2; mm = 8
        bun = sd_capsule(x, y, z, FGX + 0.10, PLAT + 1.47, FGZ - 0.01, FGX + 0.19, PLAT + 1.50, FGZ - 0.01, 0.062)
        b2 = math.sqrt((x - FGX - 0.065) ** 2 + (y - PLAT - 1.40) ** 2 + (z - FGZ + 0.01) ** 2) - 0.055
        if b2 < bun: bun = b2
        if bun < dd: dd = bun; mm = 9
        if dd < d: d = dd; m = mm
    else:
        bd = sd_box(x - FGX - 0.3, y - 1.45, z - FGZ, 1.1, 1.45, 0.7)
        if bd < d: d = max(bd, 0.03)
    # ---- undergrowth on the bank and under the trees
    if y < 2.6 and z > 2.0:
        ci = math.floor(x / 1.05); ck = math.floor(z / 1.05)
        du = 1e9
        for a in range(-1, 2):
            for b in range(-1, 2):
                i = int(ci + a); k = int(ck + b)
                h1 = hash3(i, 7, k)
                cx = (i + 0.2 + 0.6 * hash3(i, 11, k)) * 1.05; cz = (k + 0.2 + 0.6 * hash3(i, 13, k)) * 1.05
                acx = abs(cx)
                if acx < 2.05 and cz < 12.9 and cz > 5.4: continue                # the steps and the platform are kept swept
                if cz < 5.6 and (acx < 2.6 or h1 > 0.35): continue
                if h1 > 0.72: continue
                if cz > 12.9 and acx < 3.0 and cz < 14.2 and h1 > 0.3: continue
                rad = 0.30 + 0.34 * hash3(i, 17, k)
                gy = ground_h(cx, cz)
                ex = (x - cx) / rad; ey = (y - gy - 0.55 * rad) / (rad * 0.8); ez = (z - cz) / rad
                dd = (math.sqrt(ex * ex + ey * ey + ez * ez) - 1.0) * rad * 0.8
                if dd < du: du = dd
        if du < 0.35:
            du += 0.10 - 0.30 * fbm3(x * 5.0, y * 5.0, z * 5.0, 3)
        du *= 0.6
        if du < d: d = du; m = 6
    # ---- the canopy
    if y > 2.6:
        CS = 3.3
        ci = math.floor(x / CS); ck = math.floor(z / CS)
        dc = 1e9
        for a in range(-1, 2):
            for b in range(-1, 2):
                i = int(ci + a); k = int(ck + b)
                h1 = hash3(i, 23, k)
                if h1 > 0.80: continue
                cx = (i + 0.5 + 0.7 * (hash3(i, 29, k) - 0.5)) * CS; cz = (k + 0.5 + 0.7 * (hash3(i, 31, k) - 0.5)) * CS
                cy = 5.6 + 5.5 * hash3(i, 37, k)
                if cz < 9.0: cy = 6.6 + 3.0 * hash3(i, 37, k)
                vx = cx - CAMX; vy = cy - CAMY; vz = cz - CAMZ
                vl = math.sqrt(vx * vx + vy * vy + vz * vz)
                ca = (vx * GAPX + vy * GAPY + vz * GAPZ) / vl
                rad = 1.7 + 1.5 * hash3(i, 41, k)
                if ca > 0.984 and cz > 9.0: continue                               # the gap
                if ca > 0.960 and cz > 9.0: rad *= 0.62
                ex = (x - cx) / rad; ey = (y - cy) / (rad * 0.62); ez = (z - cz) / rad
                dd = (math.sqrt(ex * ex + ey * ey + ez * ez) - 1.0) * rad * 0.62
                if dd < dc: dc = dd
        if dc < 1.3:
            dc += 0.42 - 1.15 * fbm3(x * 1.5, y * 1.5, z * 1.5, 5)
        else:
            dc -= 0.75
        dc *= 0.45
        if dc < d: d = dc; m = 11
    else:
        dc = (2.75 - y)
        if dc < d and dc > 0.05: d = max(dc, 0.05)
    return d, m


# ---- the tracer: nearest surface, its normal, how open it is to the sky, and what each light can see of it
@njit(cache=True, fastmath=True, error_model='numpy')
def _soft(ox, oy, oz, dx, dy, dz, maxt, k):
    res = 1.0; t = 0.03
    for i in range(70):
        h, m = scene(ox + dx * t, oy + dy * t, oz + dz * t)
        if h < 0.0006: return 0.0
        res = min(res, k * h / t)
        t += min(max(h, 0.012), 0.6)
        if t > maxt: break
    return max(res, 0.0)


@njit(cache=True, fastmath=True, parallel=True, error_model='numpy')
def trace(O, D, tmax, L1, LP):
    n = O.shape[0]
    T = np.full(n, 1e9, np.float32); M = np.zeros(n, np.int32); Nn = np.zeros((n, 3), np.float32)
    S1 = np.zeros(n, np.float32); S2 = np.zeros(n, np.float32); AO = np.ones(n, np.float32)
    for i in prange(n):
        ox = O[i, 0]; oy = O[i, 1]; oz = O[i, 2]; dx = D[i, 0]; dy = D[i, 1]; dz = D[i, 2]
        t = 0.02; hit = False; mm = 0
        for s in range(320):
            d, m = scene(ox + dx * t, oy + dy * t, oz + dz * t)
            if d < 0.0007 * t + 0.0003:
                hit = True; mm = m; break
            t += d
            if t > tmax: break
        if not hit: continue
        px = ox + dx * t; py = oy + dy * t; pz = oz + dz * t
        e = 0.0008 * t + 0.0012
        nx = scene(px + e, py, pz)[0] - scene(px - e, py, pz)[0]
        ny = scene(px, py + e, pz)[0] - scene(px, py - e, pz)[0]
        nz = scene(px, py, pz + e)[0] - scene(px, py, pz - e)[0]
        nl = math.sqrt(nx * nx + ny * ny + nz * nz) + 1e-12
        nx /= nl; ny /= nl; nz /= nl
        T[i] = t; M[i] = mm; Nn[i, 0] = nx; Nn[i, 1] = ny; Nn[i, 2] = nz
        occ = 0.0; sca = 1.0
        for k in range(5):
            hh = 0.03 + 0.13 * k
            dd = scene(px + nx * hh, py + ny * hh, pz + nz * hh)[0]
            occ += (hh - dd) * sca; sca *= 0.75
        AO[i] = min(max(1.0 - 1.5 * occ, 0.0), 1.0)
        qx = px + nx * 0.02; qy = py + ny * 0.02; qz = pz + nz * 0.02
        S1[i] = _soft(qx, qy, qz, L1[0], L1[1], L1[2], 45.0, 7.0)
        lx = LP[0] - qx; ly = LP[1] - qy; lz = LP[2] - qz
        ll = math.sqrt(lx * lx + ly * ly + lz * lz)
        S2[i] = _soft(qx, qy, qz, lx / ll, ly / ll, lz / ll, ll - 0.05, 18.0)
    return T, M, Nn, S1, S2, AO


L1 = np.array([GAPX, GAPY, GAPZ]) / np.linalg.norm([GAPX, GAPY, GAPZ])
LP = np.array([FLX, FLY, FLZ])
LAMP_C = np.array([1.00, 0.50, 0.14])
GLOW_C = np.array([1.00, 0.74, 0.50])
SKY_C = np.array([0.15, 0.22, 0.47])


def sky(D, tile):
    """the dusk sky in any direction."""
    e = D[..., 1]
    c = (D * L1).sum(-1)
    az = np.arctan2(D[..., 0], D[..., 2]) - math.atan2(L1[0], L1[2])
    g = np.exp(-((az / 0.55) ** 2 + ((e - 0.14) / 0.17) ** 2))
    col = lerp(np.array([0.24, 0.30, 0.58]), np.array([0.07, 0.10, 0.30]), smooth(0.05, 0.75, e)[..., None])
    col = lerp(col, np.array([0.78, 0.48, 0.50]), (0.75 * smooth(0.15, 0.75, g))[..., None])
    col = lerp(col, np.array([1.00, 0.84, 0.50]), (smooth(0.45, 0.95, g) * smooth(0.36, 0.10, e))[..., None])
    # low bars of rain cloud across the afterglow
    u = az * 2.2; v = e * 9.0
    bars = map_coordinates(tile, [(v * 60.0) % 1024, (u * 22.0) % 1024], order=1, mode='wrap')
    col = lerp(col, np.array([0.40, 0.30, 0.50]), (smooth(0.52, 0.72, bars) * 0.6 * smooth(0.2, 0.8, g))[..., None])
    col = lerp(col, np.array([0.10, 0.11, 0.30]), (smooth(0.50, 0.75, bars) * 0.6 * (1 - smooth(0.1, 0.6, g)))[..., None])
    return col


def shade(P3, Nn, M, S1, S2, AO, D, T, tile):
    """what colour each visible point is, under this light."""
    def wn(a, b, scale):
        return map_coordinates(tile, [(b / scale * 97.0) % 1024, (a / scale * 97.0) % 1024], order=1, mode='wrap')
    x, y, z = P3[..., 0], P3[..., 1], P3[..., 2]
    n = len(x)
    alb = np.zeros((n, 3), np.float32); wet = np.zeros(n, np.float32); trans = np.zeros(n, np.float32)
    n1 = wn(x + 0.31 * y, z + 0.17 * y, 0.9); n2 = wn(x * 3.1 + y, z * 3.1 - y, 0.5); n3 = wn(x + z, y * 1.0 + 0.3 * z, 2.5)
    up = Nn[..., 1]
    # earth
    k = M == 1
    c = lerp(np.array([0.20, 0.085, 0.065]), np.array([0.36, 0.15, 0.09]), n1[..., None])
    leaf = smooth(0.74, 0.80, n2) * smooth(0.5, 0.8, n3)
    c = lerp(c, np.array([0.50, 0.36, 0.10]), (leaf * 0.7)[..., None])
    c = lerp(c, np.array([0.10, 0.17, 0.07]), (smooth(0.55, 0.8, n3) * smooth(5.5, 8.0, z) * 0.7)[..., None])
    alb[k] = c[k]; wet[k] = 0.5
    # laterite
    k = M == 2
    c = lerp(np.array([0.42, 0.15, 0.10]), np.array([0.64, 0.27, 0.15]), n2[..., None])
    c = c * (0.8 + 0.4 * n1[..., None])
    moss = smooth(0.50, 0.72, n3 + 0.25 * (1 - up) + 0.3 * smooth(1.2, 1.6, np.abs(x)))
    c = lerp(c, np.array([0.10, 0.19, 0.07]), (moss * 0.75)[..., None])
    alb[k] = c[k]; wet[k] = np.where(up[k] > 0.7, 0.9, 0.35)
    # bark
    k = M == 3
    c = lerp(np.array([0.10, 0.07, 0.07]), np.array([0.26, 0.17, 0.13]), n2[..., None])
    c = lerp(c, np.array([0.09, 0.17, 0.07]), (smooth(0.45, 0.75, n3) * smooth(3.0, 0.3, y - 0.9 * (z > 12)) * 0.7)[..., None])
    alb[k] = c[k]; wet[k] = 0.3
    # stone, with turmeric rubbed on by many hands and a few red flowers left at the foot
    k = M == 4
    c = lerp(np.array([0.15, 0.15, 0.18]), np.array([0.27, 0.26, 0.28]), n2[..., None])
    tur = (0.55 + 0.45 * smooth(0.3, 0.7, n1)) * smooth(PLAT + 0.12, PLAT + 0.55, y) * (z < STONEZ + 0.02)
    c = lerp(c, np.array([0.80, 0.56, 0.12]), (tur * 0.8)[..., None])
    c = lerp(c, np.array([0.70, 0.08, 0.07]), (smooth(0.70, 0.76, n2) * smooth(PLAT + 0.16, PLAT + 0.04, y) * 0.9)[..., None])
    alb[k] = c[k]; wet[k] = 0.5
    k = M == 5
    alb[k] = lerp(np.array([0.20, 0.18, 0.19]), np.array([0.30, 0.27, 0.26]), n2[..., None])[k]; wet[k] = 0.4
    # leaves
    k = (M == 6) | (M == 11)
    c = lerp(np.array([0.05, 0.16, 0.07]), np.array([0.13, 0.30, 0.09]), n2[..., None])
    c = lerp(c, np.array([0.22, 0.30, 0.06]), (smooth(0.6, 0.9, n1) * 0.5)[..., None])
    alb[k] = c[k]; wet[k] = 0.55; trans[k] = 1.0
    # her cloth: white cotton, a narrow gold line above the hem
    k = M == 7
    c = np.tile(np.array([0.90, 0.88, 0.82], np.float32), (n, 1))
    band = smooth(PLAT + 0.050, PLAT + 0.058, y) * smooth(PLAT + 0.098, PLAT + 0.090, y)
    c = lerp(c, np.array([0.80, 0.56, 0.14]), band[..., None])
    alb[k] = c[k]; trans[k] = 0.35
    k = M == 8; alb[k] = np.array([0.42, 0.24, 0.16])
    k = M == 9; alb[k] = np.array([0.50, 0.50, 0.54])
    k = M == 10; alb[k] = np.array([0.02, 0.025, 0.04]); wet[k] = 1.0

    # --- light
    V = -D
    ndl1 = (Nn * L1).sum(-1)
    lv = LP - P3; ld = np.linalg.norm(lv, axis=-1); lv = lv / ld[..., None]
    ndl2 = (Nn * lv).sum(-1)
    lamp_i = 6.5 / (ld + 0.5)                      # a painter's lamp: it reaches further than a real one
    amb = (SKY_C * (0.45 + 0.55 * up[..., None]) + np.array([0.05, 0.03, 0.03]) * (1 - up[..., None])) * AO[..., None] * 2.6
    dif1 = np.clip(ndl1, 0, 1) * S1
    dif2 = np.clip(ndl2 * 0.9 + 0.1, 0, 1) * S2 * lamp_i
    # light coming through leaf and thin cloth from behind
    th1 = np.clip(-ndl1 * 0.8 + 0.2, 0, 1) * S1 * trans
    th2 = np.clip(-ndl2, 0, 1) * S2 * lamp_i * trans * 0.6
    col = alb * (amb + GLOW_C * 0.80 * dif1[..., None] + LAMP_C * dif2[..., None])
    col += alb * np.array([1.2, 1.5, 0.5]) * (GLOW_C * 0.30 * th1[..., None]) + alb * LAMP_C * th2[..., None]
    # wet: every surface carries a little of the lamp and a little of the sky
    hv = lv + V; hv /= np.linalg.norm(hv, axis=-1, keepdims=True)
    sp = np.clip((Nn * hv).sum(-1), 0, 1)
    col += (wet * (sp ** 30 * 0.5 + sp ** 7 * 0.22) * S2 * lamp_i)[..., None] * LAMP_C
    R = D - 2 * (D * Nn).sum(-1, keepdims=True) * Nn
    fres = 0.04 + 0.96 * (1 - np.clip((V * Nn).sum(-1), 0, 1)) ** 4
    Rs = R.copy(); Rs[..., 1] = np.abs(Rs[..., 1])
    col += (wet * fres * (0.35 + 0.65 * AO) * 0.75)[..., None] * sky(Rs, tile)
    return col, fres


def draw(P):
    W, H, S, rng = P.W, P.H, P.S, P.rng
    fq = np.hypot(*np.meshgrid(np.fft.fftfreq(1024), np.fft.fftfreq(1024)))
    spec = np.fft.fft2(np.random.default_rng(SEED + 1).standard_normal((1024, 1024))) * np.where(fq > 0, 1.0 / np.maximum(fq, 1 / 200.0) ** 1.4, 0) * (fq < 0.3)
    tile = np.real(np.fft.ifft2(spec)); tile = np.clip(0.5 + 0.22 * (tile - tile.mean()) / tile.std(), 0, 1).astype(np.float32)

    RW = min(W, 1200); RH = int(RW * H / W)
    fwd = np.array([-0.2, 2.55 - CAMY, 10.0]); fwd /= np.linalg.norm(fwd)
    right = np.cross([0, 1, 0], fwd); right /= np.linalg.norm(right)
    upv = np.cross(fwd, right)
    TX, TY = 0.36, 0.45
    P.cam = (fwd, right, upv, TX, TY)
    jj, ii = np.mgrid[0:RH, 0:RW]
    nx = ((ii + 0.5) / RW * 2 - 1) * TX; ny = (1 - (jj + 0.5) / RH * 2) * TY
    D = fwd[None, None, :] + nx[..., None] * right + ny[..., None] * upv
    D /= np.linalg.norm(D, axis=-1, keepdims=True)
    D = D.reshape(-1, 3)
    O = np.tile(np.array([CAMX, CAMY, CAMZ]), (len(D), 1))
    t0 = time.time()
    T, M, Nn, S1, S2, AO = trace(O, D, 90.0, L1, LP)
    print(f"  traced {time.time() - t0:.1f}s", flush=True)
    hit = M > 0
    P3 = O + D * np.minimum(T, 90.0)[:, None]
    skyc = sky(D, tile)
    col = skyc.copy()
    c, fres = shade(P3[hit], Nn[hit], M[hit], S1[hit], S2[hit], AO[hit], D[hit], T[hit], tile)
    col[hit] = c
    # the puddle: it shows what is above it
    wk = np.nonzero(M == 10)[0]
    if len(wk):
        rip = 0.035 * (map_coordinates(tile, [(P3[wk, 2] * 260.0) % 1024, (P3[wk, 0] * 90.0) % 1024], order=1, mode='wrap') - 0.5)
        Nw = np.stack([rip, np.ones(len(wk)), rip * 2.0], 1); Nw /= np.linalg.norm(Nw, axis=1, keepdims=True)
        Dr = D[wk] - 2 * (D[wk] * Nw).sum(-1, keepdims=True) * Nw
        Or = P3[wk] + Nw * 0.01
        T2, M2, N2, S12, S22, AO2 = trace(Or, Dr, 90.0, L1, LP)
        rc = sky(Dr, tile)
        h2 = M2 > 0
        if h2.any():
            c2, _ = shade((Or + Dr * T2[:, None])[h2], N2[h2], M2[h2], S12[h2], S22[h2], AO2[h2], Dr[h2], T2[h2], tile)
            f2 = 1 - np.exp(-T2[h2] * 0.030)
            rc[h2] = lerp(c2, np.array([0.10, 0.14, 0.30]) + 0.45 * rc[h2], f2[..., None])
        # the flame itself, in the water
        fv = LP - Or; fl = np.linalg.norm(fv, axis=1); fv /= fl[:, None]
        rc += (np.clip((Dr * fv).sum(-1), 0, 1) ** 2500 * 6.0 + np.clip((Dr * fv).sum(-1), 0, 1) ** 220 * 0.8)[:, None] * LAMP_C
        fw = 0.30 + 0.70 * (1 - np.clip(-(D[wk] * Nw).sum(-1), 0, 1)) ** 3
        col[wk] = np.array([0.02, 0.025, 0.045]) + rc * fw[:, None]
    # rain-mist: distance takes things away, toward the colour of the sky behind them
    f = 1 - np.exp(-np.minimum(T, 90.0) * 0.030)
    f = np.where(hit, f, 0.0) * np.where(M == 10, 0.4, 1.0)
    mist = np.array([0.09, 0.13, 0.29]) * 0.55 + 0.50 * skyc
    col = lerp(col, mist, f[..., None])
    # the lamp in wet air
    tc = np.clip(((LP - O) * D).sum(-1), 0, np.minimum(T, 90.0))
    dmin = np.linalg.norm(O + D * tc[:, None] - LP, axis=-1)
    col += (0.018 / (dmin ** 2 + 0.010) + 0.12 * np.exp(-dmin / 1.2))[:, None] * LAMP_C * np.array([1.0, 0.9, 0.7])
    col = 1 - np.exp(-1.75 * col)                                        # the eye adjusts
    mcol = col.mean(-1, keepdims=True)
    col = np.clip(mcol + (col - mcol) * 1.22, 0, 1)
    col = col.reshape(RH, RW, 3).astype(np.float32)

    def up(a, mode=Image.BICUBIC):
        return np.asarray(Image.fromarray(a.astype(np.float32)).resize((W, H), mode))
    P.ref[:] = np.stack([up(col[..., k]) for k in range(3)], -1).clip(0, 1)
    Mi = M.reshape(RH, RW); Tn = T.reshape(RH, RW); Nr = Nn.reshape(RH, RW, 3)
    P.reg[:] = np.asarray(Image.fromarray(Mi.astype(np.int32)).resize((W, H), Image.NEAREST))
    P.mat = P.reg.copy()
    P.depth = up(np.minimum(Tn, 90.0), Image.BILINEAR)
    # the brush follows the fall of each surface: down a trunk, along a tread, round a stone
    upw = np.array([0.0, 1.0, 0.0])
    tg = np.cross(Nr, np.cross(np.broadcast_to(upw, Nr.shape), Nr))
    flat = np.abs(Nr[..., 1]) > 0.85
    tg[flat] = right
    sxv = (tg * right).sum(-1); syv = -(tg * upv).sum(-1)
    ang = np.arctan2(syv, sxv + 1e-9)
    sw = fbm(RH, RW, 60 * RW / 2400, rng, 2) * 9.0
    leafy = (Mi == 6) | (Mi == 11)
    ang = np.where(leafy, sw, ang)
    ang = np.where(Mi == 0, 0.15 + 0.5 * (fbm(RH, RW, 300 * RW / 2400, rng, 2) - 0.5), ang)
    P.ang[:] = np.asarray(Image.fromarray(ang.astype(np.float32)).resize((W, H), Image.NEAREST))
    lf = np.ones((RH, RW), np.float32)
    lf[leafy] = 0.45; lf[Mi == 3] = 2.2; lf[Mi == 0] = 1.6; lf[Mi == 2] = 1.3; lf[Mi == 1] = 1.4; lf[Mi == 10] = 1.8
    P.lenf[:] = up(lf, Image.NEAREST)
    # where the eye should be held
    X, Y = P.X, P.Y
    def project(p):
        v = np.asarray(p) - np.array([CAMX, CAMY, CAMZ])
        zc = v @ fwd
        return (0.5 + 0.5 * (v @ right) / zc / TX) * W, (0.5 - 0.5 * (v @ upv) / zc / TY) * H
    P.project = project
    fx, fy = project(LP)
    P.flame = (fx, fy)
    dfl = np.hypot(X - fx, Y - fy) / W
    det = np.zeros((RH, RW), np.float32)
    for mid, v in ((7, 1.0), (8, 1.0), (9, 1.0), (5, 0.9), (4, 0.7), (2, 0.3)):
        det[Mi == mid] = v
    P.detail[:] = np.maximum(up(det, Image.BILINEAR), 0.8 * np.exp(-(dfl / 0.06) ** 2))
    P.soft = np.clip(smooth(12.0, 40.0, P.depth) + 0.6 * (P.mat == 0) + 0.5 * (P.mat == 11) * smooth(0.25, 0.05, lum(P.ref)), 0, 1)
    P.reg[P.edge < 14 * S] = -1


def paint(P):
    W, H, S, rng = P.W, P.H, P.S, P.rng
    ref = P.ref
    L = lum(ref)
    t0 = time.time()
    say = lambda s, n: print(f"  {s:<14s}{n:>7d} strokes  {time.time() - t0:5.1f}s", flush=True)

    # 1. underpainting: the big shapes, thin, a little lower in key and greyer
    dead = blur(ref, 30 * S)
    m = dead.mean(-1, keepdims=True)
    dead = (m + (dead - m) * 0.7) * 0.92
    say("underpaint", P.layer(dead, 60, spacing=0.7, order='random', maxlen=5, tol=0.6, cjit=0.05, streak=0.04,
                              opac=0.85, thick=0.2, thickv=0.1, dry=0.6, drysens=0.6, pickup=0.0, wander=0.25))
    # 2. block-in with a broad flat brush, edge to edge
    say("block-in", P.layer(blur(ref, 10 * S), 30, spacing=0.62, maxlen=4.5, tol=0.30, cjit=0.07, streak=0.05,
                            opac=0.96, thick=0.5, thickv=0.4, dry=0.35, drysens=0.3, pickup=0.12, wander=0.18))
    P.blend(0.35 + 0.65 * P.soft, length=22, soften=5)
    # 3. modelling
    say("modelling", P.layer(blur(ref, 4.5 * S), 15, spacing=0.7, err=0.085, maxlen=4.5, tol=0.22, cjit=0.07, streak=0.05,
                             opac=0.96, thick=0.6, thickv=0.5, dry=0.4, drysens=0.3, pickup=0.14, wander=0.14))
    P.blend(0.15 + 0.6 * P.soft, length=10, soften=2.0)
    say("drawing", P.layer(blur(ref, 2.0 * S), 7.5, spacing=0.75, err=0.11, maxlen=5, tol=0.17, cjit=0.05, streak=0.04,
                           opac=0.97, thick=0.6, thickv=0.5, dry=0.4, drysens=0.3, pickup=0.10, wander=0.10))
    say("small", P.layer(blur(ref, 1.0 * S), 3.6, spacing=0.8, err=0.16, mask=0.25 + 0.75 * smooth(0.1, 0.5, P.detail),
                         maxlen=5, tol=0.14, cjit=0.035, streak=0.03, opac=0.98, thick=0.5, dry=0.35, drysens=0.25,
                         pickup=0.06, wander=0.08))
    say("accents", P.layer(ref, 1.9, spacing=0.85, err=0.20, mask=smooth(0.35, 0.7, P.detail), maxlen=5, tol=0.12,
                           cjit=0.02, streak=0.02, opac=0.99, thick=0.45, dry=0.3, drysens=0.2, pickup=0.03, wander=0.05))
    P.blend(0.5 * P.soft * smooth(0.5, 0.2, P.detail), length=6, soften=1.2)

    # 4. glaze: prussian, thin, into every dark
    P.glaze(blur(smooth(0.20, 0.03, L), 8 * S), [0.70, 0.84, 1.0], 0.45)

    # 5. the lights, loaded, laid once
    warm = ref[..., 0] - ref[..., 2]
    solid = np.isin(P.mat, (1, 2, 4, 5, 7, 8)).astype(np.float32)                 # lights go on things, not on air or leaves
    lm = np.clip(smooth(0.40, 0.70, L) + smooth(0.30, 0.55, warm) * smooth(0.22, 0.45, L), 0, 1) * solid
    lit = np.clip(ref * np.array([1.05, 1.01, 0.95]) + 0.01, 0, 1)
    say("lights", P.layer(lit, 6.0, spacing=1.0, mask=lm * 0.55, maxlen=4, tol=0.14, cjit=0.05, streak=0.05, opac=0.97,
                          thick=1.5, thickv=0.4, dry=0.3, drysens=0.25, pickup=0.04, ridge=0.6))
    say("knife", P.layer(np.clip(lit * 1.04 + 0.02, 0, 1), 4.5, spacing=1.5, mask=smooth(0.66, 0.88, L) * 0.5 * solid, maxlen=6,
                         tol=0.16, cjit=0.02, streak=0.07, opac=1.0, thick=2.2, thickv=0.2, dry=0.5, drysens=0.3,
                         pickup=0.0, edge=0.08, ridge=1.0, wander=0.04, sizev=0.5))

    fx, fy = P.flame
    # 7. rain: the first of it, slanting in from the south-west
    for i in range(700):
        x = rng.uniform(0.03, 0.97) * W; y = rng.uniform(0.03, 1.15) * W
        ln = rng.uniform(0.03, 0.075) * W
        a = 1.80 + rng.normal(0, 0.03)
        d = math.hypot(x - fx, y - fy) / W
        b = L[int(min(y, H - 1)), int(x)]
        c = np.array([0.36, 0.44, 0.64]) * (0.55 + 1.2 * b) + np.array([1.0, 0.56, 0.18]) * 0.9 * math.exp(-d / 0.09)
        P.dab(x, y, a, ln, rng.uniform(0.9, 1.6) * S, np.clip(c, 0, 1), opac=min(0.8, 0.14 + 0.5 * b + 0.55 * math.exp(-d / 0.10)),
              thick=0.35, dry=0.8, drysens=0.6, edge=0.6, streak=0.0)

    # 8. the lip of the lamp, then the flame, last
    cx_, cy_ = P.project((FLX, PLAT + 0.975, FLZ))
    rw = 0.0075 * W
    P.stroke([(cx_ - rw, cy_), (cx_, cy_ + 0.22 * rw), (cx_ + rw, cy_)], 1.6 * S, [1.0, 0.62, 0.20], thick=1.6, dry=0.2, drysens=0.1, edge=0.3)
    P.stroke([(cx_ - 0.7 * rw, cy_ + 0.5 * rw), (cx_ + 0.1 * rw, cy_ + 2.6 * rw)], 1.8 * S, [0.62, 0.26, 0.10], thick=0.8, dry=0.5, drysens=0.3, edge=0.4, opac=0.8)
    hf = 0.0185 * W
    cy = fy + 0.15 * hf
    P.stroke([(fx, cy + 0.55 * hf), (fx - 0.10 * hf, cy), (fx + 0.04 * hf, cy - 0.80 * hf)], 0.40 * hf, [1.0, 0.48, 0.07],
             thick=1.6, dry=0.2, drysens=0.1, streak=0.03, edge=0.5)
    P.stroke([(fx, cy + 0.50 * hf), (fx - 0.04 * hf, cy + 0.05 * hf), (fx + 0.02 * hf, cy - 0.52 * hf)], 0.27 * hf, [1.0, 0.80, 0.25],
             thick=2.2, dry=0.2, drysens=0.1, streak=0.03, edge=0.4)
    P.stroke([(fx, cy + 0.42 * hf), (fx, cy - 0.12 * hf)], 0.15 * hf, [1.0, 0.97, 0.80], thick=3.0, dry=0.1, drysens=0.0,
             streak=0.0, edge=0.3)
    P.sign(W - 70 * S, H - 62 * S, 30 * S)
